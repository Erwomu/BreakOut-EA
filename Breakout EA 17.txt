//| Persistent Lock Structure with GlobalVariable Storage             |
//+------------------------------------------------------------------+
struct PersistentBreakoutLock
{
    string symbol;
    string storageKey;        // GlobalVariable key
    datetime lockedD1Time;    // Current lock D1 time (in memory)
    bool isLocked;            // Memory cache flag
    
    void Init(const string sym)
    {
        symbol = sym;
        storageKey = "BO_Lock_" + sym;  // Unique key per symbol
        lockedD1Time = 0;
        isLocked = false;
        
        // ? RECOVERY: Check storage on init (survives restart)
        RecoverFromStorage();
    }
    
    // Recover lock state from persistent storage
    void RecoverFromStorage()
    {
        datetime storedTime = (datetime)GlobalVariableGet(storageKey);
        
        if(storedTime > 0)
        {
            datetime currentD1 = iTime(symbol, PERIOD_D1, 0);
            
            // If stored time matches current D1, restore lock
            if(storedTime == currentD1)
            {
                lockedD1Time = storedTime;
                isLocked = true;
                
                Print("¨X¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨[");
                PrintFormat("¨U  ? %s: LOCK RECOVERED", symbol);
                PrintFormat("¨U     From: GlobalVariable storage");
                PrintFormat("¨U     Date: %s", TimeToString(storedTime, TIME_DATE));
                PrintFormat("¨U     Status: LOCKED (no trading today)");
                Print("¨^¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨a");
            }
            else
            {
                // Storage is outdated (old D1), clear it
                GlobalVariableDel(storageKey);
                PrintFormat("? %s: Old lock cleared (D1 changed)", symbol);
            }
        }
        else
        {
            PrintFormat("? %s: No previous lock found (ready to trade)", symbol);
        }
    }
    
    // Check if we can trade this symbol today
    bool CanTrade()
    {
        datetime currentD1 = iTime(symbol, PERIOD_D1, 0);
        
        // ? D1 candle changed? Auto-unlock
        if(currentD1 != lockedD1Time && lockedD1Time > 0)
        {
            Print("¨X¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨[");
            PrintFormat("¨U  ? %s: NEW D1 CANDLE", symbol);
            PrintFormat("¨U     Previous: %s", TimeToString(lockedD1Time, TIME_DATE));
            PrintFormat("¨U     Current:  %s", TimeToString(currentD1, TIME_DATE));
            PrintFormat("¨U     Status: UNLOCKED (ready for new trade)");
            Print("¨^¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨a");
            Unlock();
            return true;
        }
        
        // Check memory cache first (fastest)
        if(isLocked)
        {
            // Occasional log to confirm lock is active
            static datetime lastLog = 0;
            if(TimeCurrent() - lastLog > 3600) // Log once per hour
            {
                PrintFormat("? %s: Still locked for %s (checked from cache)", 
                           symbol, TimeToString(lockedD1Time, TIME_DATE));
                lastLog = TimeCurrent();
            }
            return false;
        }
        
        // Double-check storage (in case of memory issues)
        datetime storedTime = (datetime)GlobalVariableGet(storageKey);
        if(storedTime == currentD1 && storedTime > 0)
        {
            // Storage says locked but memory didn't - sync them
            lockedD1Time = storedTime;
            isLocked = true;
            
            PrintFormat("?? %s: Lock state synced from storage", symbol);
            return false;
        }
        
        return true;
    }
    
    // Lock the symbol for current D1 period
    void Lock()
    {
        datetime currentD1 = iTime(symbol, PERIOD_D1, 0);
        
        // Update memory
        lockedD1Time = currentD1;
        isLocked = true;
        
        // ? PERSIST to storage (survives restart)
        if(!GlobalVariableSet(storageKey, (double)currentD1))
        {
            PrintFormat("?? WARNING: Failed to save lock to storage for %s", symbol);
        }
        
        // Verify storage write
        datetime stored = (datetime)GlobalVariableGet(storageKey);
        bool storageOK = (stored == currentD1);
        
        Print("¨X¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨[");
        PrintFormat("¨U  ? %s LOCKED FOR TODAY", symbol);
        PrintFormat("¨U     Lock Time: %s", TimeToString(currentD1, TIME_DATE|TIME_MINUTES));
        PrintFormat("¨U     D1 Candle: %s", TimeToString(currentD1, TIME_DATE));
        PrintFormat("¨U     Storage Key: %s", storageKey);
        PrintFormat("¨U     Storage Status: %s", storageOK ? "? SAVED" : "? FAILED");
        PrintFormat("¨U     Memory Status: ? LOCKED");
        PrintFormat("¨U");
        PrintFormat("¨U     ? NO MORE PROCESSING TODAY");
        PrintFormat("¨U     ? Will unlock at next D1 candle");
        Print("¨^¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨a");
    }
    
    // Unlock (manual or automatic)
    void Unlock()
    {
        isLocked = false;
        lockedD1Time = 0;
        
        // Clear from storage
        if(GlobalVariableCheck(storageKey))
        {
            GlobalVariableDel(storageKey);
        }
        
        PrintFormat("? %s: UNLOCKED (ready for new trade)", symbol);
    }
    
    // Get lock status for dashboard
    string GetStatus()
    {
        if(isLocked)
        {
            return StringFormat("? LOCKED (%s)", TimeToString(lockedD1Time, TIME_DATE));
        }
        return "? READY";
    }
};

//+------------------------------------------------------------------+
//| Check ADX Filter                                                 |
//+------------------------------------------------------------------+
bool CheckADXFilter(const string symbol, const int index)
{
    if(!UseADXFilter)
        return true;
    
    double adxValue = GetADXValue(index);
    
    if(adxValue <= 0)
    {
        PrintFormat("WARNING: %s - Invalid ADX value", symbol);
        return false;
    }
    
    if(adxValue < MinADX)
    {
        PrintFormat("%s: ADX Filter REJECTED | ADX: %.2f < Min: %.2f", 
                   symbol, adxValue, MinADX);
        return false;
    }
    
    PrintFormat("%s: ADX Filter PASSED | ADX: %.2f >= Min: %.2f", 
               symbol, adxValue, MinADX);
    return true;
}

//+------------------------------------------------------------------+
//| Price Position Detection                                          |
//+------------------------------------------------------------------+
RejectionSignal PricePosition(const string symbol, double keyLevel)
{
    RejectionSignal signal;
    signal.Reset();
    
    double open[], close[], high[], low[];
    datetime time[];
    
    if(CopyOpen(symbol, PERIOD_H1, 1, 3, open) < 3 ||
       CopyClose(symbol, PERIOD_H1, 1, 3, close) < 3 ||
       CopyHigh(symbol, PERIOD_H1, 1, 3, high) < 3 ||
       CopyLow(symbol, PERIOD_H1, 1, 3, low) < 3 ||
       CopyTime(symbol, PERIOD_H1, 1, 3, time) < 3)
    {
        Print("ERROR: Failed to copy price data for ", symbol);
        return signal;
    }
    
    double o1 = open[0],  c1 = close[0],  h1 = high[0],  l1 = low[0];
    double o2 = open[1],  c2 = close[1],  h2 = high[1],  l2 = low[1];
    
    signal.signalTime = time[0];
    
    double body1 = MathAbs(c1 - o1);
    double upperWick1 = h1 - MathMax(o1, c1);
    double lowerWick1 = MathMin(o1, c1) - l1;
    double totalRange1 = h1 - l1;
    
    if(totalRange1 < 0.00001)
        return signal;
    
    double bodyRatio = body1 / totalRange1;
    double upperWickRatio = upperWick1 / totalRange1;
    double lowerWickRatio = lowerWick1 / totalRange1;
    
    if(bodyRatio < 0.25)
        return signal;
    
    // BULLISH PATTERNS
    if(lowerWickRatio > 0.5 && bodyRatio < 0.4)
    {
        signal.isBullishRejection = true;
        signal.rejectionType = "PIN_BAR";
    }
    else if(c1 > o1 && o2 > c2 && c1 > o2 && o1 >= c2)
    {
        signal.isBullishRejection = true;
        signal.rejectionType = "ENGULFING";
    }
    else if(h1 <= h2 && l1 >= l2 && c1 > o1)
    {
        signal.isBullishRejection = true;
        signal.rejectionType = "INSIDE_BAR";
    }
    else if(c1 > o1 && o2 > c2 && c1 < o2 && o1 > c2)
    {
        signal.isBullishRejection = true;
        signal.rejectionType = "HARAMI";
    }
    
    // BEARISH PATTERNS
    if(upperWickRatio > 0.5 && bodyRatio < 0.4)
    {
        signal.isBearishRejection = true;
        signal.rejectionType = "SHOOTING_STAR";
    }
    else if(c1 < o1 && o2 < c2 && c1 < o2 && o1 <= c2)
    {
        signal.isBearishRejection = true;
        signal.rejectionType = "ENGULFING";
    }
    else if(h1 <= h2 && l1 >= l2 && c1 < o1)
    {
        signal.isBearishRejection = true;
        signal.rejectionType = "INSIDE_BAR";
    }
    else if(c1 < o1 && o2 < c2 && c1 > o2 && o1 < c2)
    {
        signal.isBearishRejection = true;
        signal.rejectionType = "HARAMI";
    }
    
    return signal;
}

//+------------------------------------------------------------------+
//| Calculate Dynamic Risk Based on Account State                    |
//+------------------------------------------------------------------+
double CalculateDynamicRiskPercentage()
{
    if(!UseAdvancedRisk)
        return RiskPercent;
    
    double currentBalance = AccountInfoDouble(ACCOUNT_BALANCE);
    
    if(currentBalance > g_advPeakBalance)
        g_advPeakBalance = currentBalance;
    
    if(g_advInitialBalance > 0)
        g_advCurrentProfit = ((currentBalance - g_advInitialBalance) / g_advInitialBalance) * 100.0;
    
    if(g_advPeakBalance > 0)
        g_advCurrentDrawdown = ((g_advPeakBalance - currentBalance) / g_advPeakBalance) * 100.0;
    
    double adjustedRisk = RiskPercent;
    
    if(UseDrawdownDefense && g_advCurrentDrawdown >= DrawdownThreshold)
    {
        adjustedRisk *= DrawdownRiskMultiplier;
        
        static datetime lastDrawdownAlert = 0;
        if(TimeCurrent() - lastDrawdownAlert > 3600)
        {
            PrintFormat("?? DRAWDOWN DEFENSE: %.2f%% DD | Risk: %.2f%% Â¡Ãº %.2f%%", 
                       g_advCurrentDrawdown, RiskPercent, adjustedRisk);
            lastDrawdownAlert = TimeCurrent();
        }
    }
    else if(UseProfitAccelerator && g_advCurrentProfit >= ProfitThreshold)
    {
        adjustedRisk *= ProfitRiskMultiplier;
        
        static datetime lastProfitAlert = 0;
        if(TimeCurrent() - lastProfitAlert > 3600)
        {
            PrintFormat("? PROFIT ACCELERATOR: +%.2f%% | Risk: %.2f%% Â¡Ãº %.2f%%", 
                       g_advCurrentProfit, RiskPercent, adjustedRisk);
            lastProfitAlert = TimeCurrent();
        }
    }
    
    adjustedRisk = MathMin(adjustedRisk, MaxRiskPercent);
    adjustedRisk = MathMax(adjustedRisk, 0.1);
    
    g_advAdjustedRisk = adjustedRisk;
    return adjustedRisk;
}

//+------------------------------------------------------------------+
//| Get Advanced Risk Status                                         |
//+------------------------------------------------------------------+
string GetAdvancedRiskStatus()
{
    if(!UseAdvancedRisk)
        return "";
        
    string status = "";
    
    if(g_advCurrentDrawdown >= DrawdownThreshold && UseDrawdownDefense)
    {
        status = StringFormat(" | ?? DEFENSE (%.1f%% DD)", g_advCurrentDrawdown);
    }
    else if(g_advCurrentProfit >= ProfitThreshold && UseProfitAccelerator)
    {
        status = StringFormat(" | ? ACCELERATOR (+%.1f%%)", g_advCurrentProfit);
    }
    
    return status;
}

//+------------------------------------------------------------------+
//| Initialize in OnInit()                                            |
//+------------------------------------------------------------------+
int OnInit()
{
    Print("========================================");
    Print("Initializing Breakout EA v2.15");
    Print("? ONE TRADE PER SYMBOL PER DAY");
    Print("? PERSISTENT LOCK (survives restarts)");
    Print("========================================");
    
    g_eaTrade.SetExpertMagicNumber(MagicNumber);
    g_eaTrade.SetDeviationInPoints(10);
    g_eaTrade.SetTypeFilling(ORDER_FILLING_IOC);
    g_eaTrade.SetAsyncMode(false);
    g_eaTrade.LogLevel(LOG_LEVEL_ERRORS);
    
    InitializeTradingSymbols();
    
    int validSymbols = 0;
    
    for(int i = 0; i < SYMBOL_COUNT; i++)
    {
        string symbol = g_tradingSymbols[i].symbol;
        
        if(!SymbolInfoInteger(symbol, SYMBOL_SELECT))
        {
            if(!SymbolSelect(symbol, true))
            {
                PrintFormat("ERROR: Symbol %s not found", symbol);
                continue;
            }
        }
        
        g_tradingSymbols[i].atrHandle = iATR(symbol, PERIOD_H1, ATRPeriod);
        if(g_tradingSymbols[i].atrHandle == INVALID_HANDLE)
        {
            PrintFormat("ERROR: Failed to create ATR handle for %s", symbol);
            continue;
        }
        
        g_tradingSymbols[i].adxHandle = iADX(symbol, PERIOD_H4, ADXPeriod);
        if(g_tradingSymbols[i].adxHandle == INVALID_HANDLE)
        {
            PrintFormat("ERROR: Failed to create ADX handle for %s", symbol);
            IndicatorRelease(g_tradingSymbols[i].atrHandle);
            continue;
        }
        
        Sleep(100);
        
        g_htfData[i].Reset();
        g_stats[i].Reset();
        
        // ? Initialize persistent lock (recovers from storage if needed)
        g_symbolLocks[i].Init(symbol);
        
        if(!MarkPreviousHTF(symbol, i))
        {
            PrintFormat("WARNING: Failed to mark HTF levels for %s", symbol);
        }
        
        g_tradingSymbols[i].isActive = true;
        validSymbols++;
        PrintFormat("? %s initialized | Lock status: %s", symbol, g_symbolLocks[i].GetStatus());
    }
    
    if(validSymbols == 0)
    {
        Print("ERROR: No valid symbols to trade");
        return INIT_FAILED;
    }
    
    PrintFormat("? %d symbol(s) ready", validSymbols);
    
    EventSetTimer(60);
    
    g_initialized = true;
    g_lastCheck = TimeCurrent();
    
    Print("========================================");
    Print("? Breakout EA initialized successfully!");
    Print("? Persistent lock system active");
    Print("? Locks survive EA restarts");
    Print("========================================");
    
    return INIT_SUCCEEDED;
}

//+------------------------------------------------------------------+
//| Clean up on exit                                                  |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
    Print("========================================");
    Print("Deinitializing Breakout EA");
    PrintFormat("Reason: %s", GetDeinitReasonText(reason));
    Print("========================================");
    
    EventKillTimer();
    
    // Note: We DON'T clear GlobalVariables on deinit
    // This allows locks to persist through restarts
    
    for(int i = 0; i < SYMBOL_COUNT; i++)
    {
        if(g_tradingSymbols[i].atrHandle != INVALID_HANDLE)
        {
            IndicatorRelease(g_tradingSymbols[i].atrHandle);
            g_tradingSymbols[i].atrHandle = INVALID_HANDLE;
        }
        
        if(g_tradingSymbols[i].adxHandle != INVALID_HANDLE)
        {
            IndicatorRelease(g_tradingSymbols[i].adxHandle);
            g_tradingSymbols[i].adxHandle = INVALID_HANDLE;
        }
        
        if(g_tradingSymbols[i].isActive)
        {
            ObjectDelete(0, "HTF_High_" + g_tradingSymbols[i].symbol);
            ObjectDelete(0, "HTF_Low_" + g_tradingSymbols[i].symbol);
            ObjectDelete(0, "HTF_Range_" + g_tradingSymbols[i].symbol);
            
            // Show final lock status
            PrintFormat("   %s: %s", 
                       g_tradingSymbols[i].symbol, 
                       g_symbolLocks[i].GetStatus());
        }
    }
    
    if(ShowDashboard)
    {
        RemoveDashboard();
    }
    
    ChartRedraw();
    
    Print("========================================");
    Print("? Lock states saved to GlobalVariables");
    Print("? Will be restored on next EA start");
    Print("========================================");
    Print("? Breakout EA deinitialized successfully");
}

//+------------------------------------------------------------------+
//| ? NEW: Check if traded today using tracker + history           |
//+------------------------------------------------------------------+
bool HasTradedToday(const string symbol, const int index)
{
    datetime currentD1 = iTime(symbol, PERIOD_D1, 0);
    
    // ? CHECK 1: In-memory tracker (fastest check)
    if(g_dailyTracker[index].tradeExecutedToday && 
       g_dailyTracker[index].lastTradeD1Time == currentD1)
    {
        PrintFormat("? %s: BLOCKED by in-memory tracker (ticket #%I64u)", 
                   symbol, g_dailyTracker[index].lastTradeTicket);
        return true;
    }
    
    // ? CHECK 2: Verify with history (in case EA restarted)
    datetime currentD1Start = currentD1;
    datetime currentD1End = currentD1Start + PeriodSeconds(PERIOD_D1);
    
    // Check open positions
    for(int i = 0; i < PositionsTotal(); i++)
    {
        if(!g_eaPosition.SelectByIndex(i))
            continue;
        
        if(g_eaPosition.Symbol() != symbol || g_eaPosition.Magic() != MagicNumber)
            continue;
        
        datetime posOpenTime = (datetime)g_eaPosition.Time();
        
        if(posOpenTime >= currentD1Start && posOpenTime < currentD1End)
        {
            // Update tracker if not set
            if(!g_dailyTracker[index].tradeExecutedToday)
            {
                g_dailyTracker[index].lastTradeD1Time = currentD1;
                g_dailyTracker[index].lastTradeTicket = g_eaPosition.Ticket();
                g_dailyTracker[index].tradeExecutedToday = true;
            }
            
            PrintFormat("? %s: BLOCKED - found open position #%I64u", 
                       symbol, g_eaPosition.Ticket());
            return true;
        }
    }
    
    // Check closed positions
    if(!HistorySelect(currentD1Start - 3600, TimeCurrent() + 3600))
        return false;
    
    int totalDeals = HistoryDealsTotal();
    
    for(int i = 0; i < totalDeals; i++)
    {
        ulong ticket = HistoryDealGetTicket(i);
        if(ticket == 0)
            continue;
        
        if(HistoryDealGetString(ticket, DEAL_SYMBOL) != symbol)
            continue;
        
        if(HistoryDealGetInteger(ticket, DEAL_MAGIC) != MagicNumber)
            continue;
        
        if(HistoryDealGetInteger(ticket, DEAL_ENTRY) != DEAL_ENTRY_IN)
            continue;
        
        datetime dealTime = (datetime)HistoryDealGetInteger(ticket, DEAL_TIME);
        
        if(dealTime >= currentD1Start && dealTime < currentD1End)
        {
            // Update tracker
            if(!g_dailyTracker[index].tradeExecutedToday)
            {
                g_dailyTracker[index].lastTradeD1Time = currentD1;
                g_dailyTracker[index].lastTradeTicket = ticket;
                g_dailyTracker[index].tradeExecutedToday = true;
            }
            
            PrintFormat("? %s: BLOCKED - found closed deal #%I64u", symbol, ticket);
            return true;
        }
    }
    
    PrintFormat("? %s: OK - No trades today, can proceed", symbol);
    return false;
}

//+------------------------------------------------------------------+
//| Calculate Lot Size with Enhanced Risk Management                |
//+------------------------------------------------------------------+
double CalculateLotSize(const string symbol, const int index, const double atr)
{
    if(!g_eaSymbolInfo.Name(symbol) || atr <= 0)
        return g_eaSymbolInfo.LotsMin();
    
    double riskPercent = CalculateDynamicRiskPercentage();
    
    double accountBalance = AccountInfoDouble(ACCOUNT_BALANCE);
    double riskAmount = (riskPercent / 100.0) * accountBalance;
    
    double stopLossDistance = atr * ATRStopLoss;
    double tickValue = g_eaSymbolInfo.TickValue();
    double tickSize = g_eaSymbolInfo.TickSize();
    
    if(tickValue <= 0 || tickSize <= 0 || stopLossDistance <= 0)
        return g_eaSymbolInfo.LotsMin();
    
    double stopLossInTicks = stopLossDistance / tickSize;
    double lotSize = riskAmount / (stopLossInTicks * tickValue);
    
    if(symbol == "XAUUSD")
        lotSize /= 10.0;
    
    double minLot = g_eaSymbolInfo.LotsMin();
    double maxLot = g_eaSymbolInfo.LotsMax();
    double lotStep = g_eaSymbolInfo.LotsStep();
    
    lotSize = MathMax(minLot, MathMin(lotSize, maxLot));
    lotSize = NormalizeDouble(MathFloor(lotSize / lotStep) * lotStep, 2);
    
    if(UseAdvancedRisk && riskPercent != RiskPercent)
    {
        PrintFormat("%s: Risk %.2f%% (Base: %.2f%%) | Lot: %.2f | $%.2f at risk",
                   symbol, riskPercent, RiskPercent, lotSize, riskAmount);
    }
    
    return lotSize;
}

//+------------------------------------------------------------------+
//| Manage Positions - Enhanced with Manual TP/SL Close              |
//+------------------------------------------------------------------+
void ManagePositions(const string symbol, const int index)
{
    double atr = GetATRValue(index);
    if(atr <= 0)
        return;
    
    for(int i = PositionsTotal() - 1; i >= 0; i--)
    {
        if(!g_eaPosition.SelectByIndex(i))
            continue;
        
        if(g_eaPosition.Symbol() != symbol || g_eaPosition.Magic() != MagicNumber)
            continue;
        
        if(!g_eaSymbolInfo.Name(symbol))
            continue;
        
        g_eaSymbolInfo.RefreshRates();
        
        double currentSL = g_eaPosition.StopLoss();
        double currentTP = g_eaPosition.TakeProfit();
        double openPrice = g_eaPosition.PriceOpen();
        int digits = (int)g_eaSymbolInfo.Digits();
        
        //+------------------------------------------------------------------+
        //| MANUAL TP CLOSE CHECK - PRIORITY #1                              |
        //+------------------------------------------------------------------+
        if(currentTP > 0)
        {
            bool closeAtTP = false;
            double tpBuffer = g_eaSymbolInfo.Point() * 3;
            
            if(g_eaPosition.PositionType() == POSITION_TYPE_BUY)
            {
                double bid = g_eaSymbolInfo.Bid();
                
                if(bid >= (currentTP - tpBuffer))
                {
                    closeAtTP = true;
                    double pipsToTP = (currentTP - bid) / g_eaSymbolInfo.Point() / 10;
                    PrintFormat("%s BUY: Manual TP close triggered at %.5f (TP: %.5f, Distance: %.1f pips)", 
                               symbol, bid, currentTP, pipsToTP);
                }
            }
            else if(g_eaPosition.PositionType() == POSITION_TYPE_SELL)
            {
                double ask = g_eaSymbolInfo.Ask();
                
                if(ask <= (currentTP + tpBuffer))
                {
                    closeAtTP = true;
                    double pipsToTP = (ask - currentTP) / g_eaSymbolInfo.Point() / 10;
                    PrintFormat("%s SELL: Manual TP close triggered at %.5f (TP: %.5f, Distance: %.1f pips)", 
                               symbol, ask, currentTP, pipsToTP);
                }
            }
            
            if(closeAtTP)
            {
                ulong ticket = g_eaPosition.Ticket();
                double profit = g_eaPosition.Profit() + g_eaPosition.Swap() + g_eaPosition.Commission();
                
                if(g_eaTrade.PositionClose(ticket))
                {
                    if(g_eaTrade.ResultRetcode() == TRADE_RETCODE_DONE)
                    {
                        PrintFormat("ÃƒÂƒÃ‚ÂƒÃƒÂ‚Ã‚Â¢ÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚Âœ? %s: Position #%I64u closed manually at TP | Profit: $%.2f", 
                                   symbol, ticket, profit);
                    }
                    else
                    {
                        PrintFormat("ÃƒÂƒÃ‚ÂƒÃƒÂ‚Ã‚Â¢ÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚ÂšÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚Â ÃƒÂƒÃ‚ÂƒÃƒÂ‚Ã‚Â¯ÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚Â¸ÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚Â %s: TP close attempt for #%I64u - Retcode: %d (%s)", 
                                   symbol, ticket, g_eaTrade.ResultRetcode(), 
                                   g_eaTrade.ResultRetcodeDescription());
                    }
                }
                else
                {
                    PrintFormat("ÃƒÂƒÃ‚ÂƒÃƒÂ‚Ã‚Â¢ÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚Â? %s: Failed to close #%I64u at TP | Error: %d", 
                               symbol, ticket, GetLastError());
                }
                
                continue;
            }
        }
        
        //+------------------------------------------------------------------+
        //| MANUAL SL CLOSE CHECK - PRIORITY #2                              |
        //+------------------------------------------------------------------+
        if(currentSL > 0)
        {
            bool closeAtSL = false;
            double slBuffer = g_eaSymbolInfo.Point() * 3;
            
            if(g_eaPosition.PositionType() == POSITION_TYPE_BUY)
            {
                double bid = g_eaSymbolInfo.Bid();
                
                if(bid <= (currentSL + slBuffer))
                {
                    closeAtSL = true;
                    double pipsFromSL = (bid - currentSL) / g_eaSymbolInfo.Point() / 10;
                    PrintFormat("%s BUY: Manual SL close triggered at %.5f (SL: %.5f, Distance: %.1f pips)", 
                               symbol, bid, currentSL, pipsFromSL);
                }
            }
            else if(g_eaPosition.PositionType() == POSITION_TYPE_SELL)
            {
                double ask = g_eaSymbolInfo.Ask();
                
                if(ask >= (currentSL - slBuffer))
                {
                    closeAtSL = true;
                    double pipsFromSL = (currentSL - ask) / g_eaSymbolInfo.Point() / 10;
                    PrintFormat("%s SELL: Manual SL close triggered at %.5f (SL: %.5f, Distance: %.1f pips)", 
                               symbol, ask, currentSL, pipsFromSL);
                }
            }
            
            if(closeAtSL)
            {
                ulong ticket = g_eaPosition.Ticket();
                double profit = g_eaPosition.Profit() + g_eaPosition.Swap() + g_eaPosition.Commission();
                
                if(g_eaTrade.PositionClose(ticket))
                {
                    if(g_eaTrade.ResultRetcode() == TRADE_RETCODE_DONE)
                    {
                        PrintFormat("ÃƒÂƒÃ‚ÂƒÃƒÂ‚Ã‚Â°ÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚ÂŸÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚Â›ÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚Â‘ %s: Position #%I64u closed manually at SL | Loss: $%.2f", 
                                   symbol, ticket, profit);
                    }
                    else
                    {
                        PrintFormat("ÃƒÂƒÃ‚ÂƒÃƒÂ‚Ã‚Â¢ÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚ÂšÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚Â ÃƒÂƒÃ‚ÂƒÃƒÂ‚Ã‚Â¯ÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚Â¸ÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚Â %s: SL close attempt for #%I64u - Retcode: %d (%s)", 
                                   symbol, ticket, g_eaTrade.ResultRetcode(), 
                                   g_eaTrade.ResultRetcodeDescription());
                    }
                }
                else
                {
                    PrintFormat("ÃƒÂƒÃ‚ÂƒÃƒÂ‚Ã‚Â¢ÃƒÂƒÃ‚Â‚ÃƒÂ‚Ã‚Â? %s: Failed to close #%I64u at SL | Error: %d", 
                               symbol, ticket, GetLastError());
                }
                
                continue;
            }
        }
        
        // Continue with trailing stop logic
        bool modified = false;
        double newSL = 0.0;
        
        if(g_eaPosition.PositionType() == POSITION_TYPE_BUY)
        {
            double bid = g_eaSymbolInfo.Bid();
            double profitDistance = bid - openPrice;
            
            if(profitDistance >= (atr * BEMultiplier) && currentSL < openPrice)
            {
                newSL = NormalizeDouble(openPrice + (g_eaSymbolInfo.Point() * 10), digits);
                modified = true;
                PrintFormat("%s BUY: Moving to Break-Even at %.5f", symbol, newSL);
            }
            else if(profitDistance >= (atr * TrailStartMultiplier) && currentSL >= openPrice)
            {
                double trailSL = NormalizeDouble(bid - (atr * TrailStopMultiplier), digits);
                if(trailSL > currentSL)
                {
                    newSL = trailSL;
                    modified = true;
                    PrintFormat("%s BUY: Trailing SL to %.5f", symbol, newSL);
                }
            }
        }
        else if(g_eaPosition.PositionType() == POSITION_TYPE_SELL)
        {
            double ask = g_eaSymbolInfo.Ask();
            double profitDistance = openPrice - ask;
            
            if(profitDistance >= (atr * BEMultiplier) && (currentSL == 0 || currentSL > openPrice))
            {
                newSL = NormalizeDouble(openPrice - (g_eaSymbolInfo.Point() * 10), digits);
                modified = true;
                PrintFormat("%s SELL: Moving to Break-Even at %.5f", symbol, newSL);
            }
            else if(profitDistance >= (atr * TrailStartMultiplier) && currentSL > 0 && currentSL <= openPrice)
            {
                double trailSL = NormalizeDouble(ask + (atr * TrailStopMultiplier), digits);
                if(currentSL == 0 || trailSL < currentSL)
                {
                    newSL = trailSL;
                    modified = true;
                    PrintFormat("%s SELL: Trailing SL to %.5f", symbol, newSL);
                }
            }
        }
        
        if(modified && newSL > 0)
        {
            if(g_eaTrade.PositionModify(g_eaPosition.Ticket(), newSL, g_eaPosition.TakeProfit()))
            {
                if(g_eaTrade.ResultRetcode() == TRADE_RETCODE_DONE)
                {
                    PrintFormat("%s: SL modified successfully to %.5f", symbol, newSL);
                }
                else
                {
                    PrintFormat("%s: SL modify returned code %d", symbol, g_eaTrade.ResultRetcode());
                }
            }
            else
            {
                PrintFormat("%s: Failed to modify SL | Error:%d", symbol, g_eaTrade.ResultRetcode());
            }
        }
    }
}

//+------------------------------------------------------------------+
//| Check Trading Session                                            |
//+------------------------------------------------------------------+
bool IsInTradingSession()
{
    MqlDateTime current;
    TimeToStruct(TimeCurrent(), current);
    int currentHour = current.hour;
    
    if(currentHour >= LondonStart && currentHour < LondonEnd)
        return true;
    
    if(currentHour >= NewYorkStart && currentHour < NewYorkEnd)
        return true;
    
    return false;
}

//+------------------------------------------------------------------+
//| Check Spread                                                     |
//+------------------------------------------------------------------+
bool IsSpreadAcceptable(const string symbol)
{
    if(!g_eaSymbolInfo.Name(symbol))
        return false;
    
    g_eaSymbolInfo.RefreshRates();
    
    double spread = g_eaSymbolInfo.Spread() * g_eaSymbolInfo.Point();
    double maxSpread = MaxSpreadPips * g_eaSymbolInfo.Point() * 10;
    
    if(spread > maxSpread)
    {
        PrintFormat("%s: Spread %.1f pips exceeds maximum %.1f pips", 
                   symbol, spread / (g_eaSymbolInfo.Point() * 10), MaxSpreadPips);
        return false;
    }
    
    return true;
}

//+------------------------------------------------------------------+
//| Update Statistics                                                |
//+------------------------------------------------------------------+
void UpdateStatistics()
{
    if(!HistorySelect(0, TimeCurrent()))
        return;
    
    for(int i = 0; i < SYMBOL_COUNT; i++)
    {
        if(!g_tradingSymbols[i].isActive)
            continue;
        
        string symbol = g_tradingSymbols[i].symbol;
        double totalProfit = 0;
        int totalTrades = 0;
        int wins = 0;
        int losses = 0;
        
        for(int j = HistoryDealsTotal() - 1; j >= 0; j--)
        {
            ulong ticket = HistoryDealGetTicket(j);
            if(ticket == 0)
                continue;
            
            if(HistoryDealGetString(ticket, DEAL_SYMBOL) != symbol)
                continue;
            
            if(HistoryDealGetInteger(ticket, DEAL_MAGIC) != MagicNumber)
                continue;
            
            if(HistoryDealGetInteger(ticket, DEAL_ENTRY) != DEAL_ENTRY_OUT)
                continue;
            
            double profit = HistoryDealGetDouble(ticket, DEAL_PROFIT);
            totalProfit += profit;
            totalTrades++;
            
            if(profit > 0)
                wins++;
            else if(profit < 0)
                losses++;
        }
        
        g_stats[i].totalProfit = totalProfit;
        g_stats[i].totalTrades = totalTrades;
        g_stats[i].winningTrades = wins;
        g_stats[i].losingTrades = losses;
    }
}

//+------------------------------------------------------------------+
//| Update Dashboard with Lock Status                                |
//+------------------------------------------------------------------+
void UpdateDashboard()
{
    double balance = AccountInfoDouble(ACCOUNT_BALANCE);
    double equity = AccountInfoDouble(ACCOUNT_EQUITY);
    double profit = equity - balance;
    int totalPos = GetAllPositionsCount();
    
    string riskStatus = GetAdvancedRiskStatus();
    
    string accountText = StringFormat("Balance: $%.2f | Equity: $%.2f | P/L: $%.2f | Pos: %d/%d%s",
                                     balance, equity, profit, totalPos, MaxTotalPositions, riskStatus);
    ObjectSetString(0, "Dashboard_Account", OBJPROP_TEXT, accountText);
    ObjectSetInteger(0, "Dashboard_Account", OBJPROP_COLOR, profit >= 0 ? clrLime : clrRed);
    
    bool inSession = IsInTradingSession();
    string sessionText = StringFormat("Session: %s | HTF: D1 | LTF: H1 | ? Persistent Locks: ON",
                                     inSession ? "ACTIVE" : "CLOSED");
    ObjectSetString(0, "Dashboard_Session", OBJPROP_TEXT, sessionText);
    ObjectSetInteger(0, "Dashboard_Session", OBJPROP_COLOR, inSession ? clrLime : clrOrange);
    
    if(UseAdvancedRisk)
    {
        string riskStateText = "";
        color riskColor = clrWhite;
        
        if(g_advCurrentDrawdown >= DrawdownThreshold && UseDrawdownDefense)
        {
            riskStateText = StringFormat("Risk: %.2f%% | ?? DEFENSE (%.1f%% DD)", 
                                       g_advAdjustedRisk, g_advCurrentDrawdown);
            riskColor = clrOrange;
        }
        else if(g_advCurrentProfit >= ProfitThreshold && UseProfitAccelerator)
        {
            riskStateText = StringFormat("Risk: %.2f%% | ? ACCELERATOR (+%.1f%%)", 
                                       g_advAdjustedRisk, g_advCurrentProfit);
            riskColor = clrLime;
        }
        else
        {
            riskStateText = StringFormat("Risk: %.2f%% | ?? NORMAL", g_advAdjustedRisk);
            riskColor = clrWhite;
        }
        
        ObjectSetString(0, "Dashboard_RiskState", OBJPROP_TEXT, riskStateText);
        ObjectSetInteger(0, "Dashboard_RiskState", OBJPROP_COLOR, riskColor);
    }
    
    // Symbol status with lock info
    for(int i = 0; i < SYMBOL_COUNT; i++)
    {
        string labelName = "Dashboard_Symbol_" + IntegerToString(i);
        
        if(!g_tradingSymbols[i].isActive)
        {
            ObjectSetString(0, labelName, OBJPROP_TEXT, "");
            continue;
        }
        
        string symbol = g_tradingSymbols[i].symbol;
        string status = "-";
        color textColor = clrGray;
        
        int positions = GetSymbolPositionsCount(symbol);
        
        if(positions > 0)
        {
            // Has open position
            double symbolProfit = 0;
            for(int j = 0; j < PositionsTotal(); j++)
            {
                if(g_eaPosition.SelectByIndex(j))
                {
                    if(g_eaPosition.Symbol() == symbol && g_eaPosition.Magic() == MagicNumber)
                    {
                        symbolProfit += g_eaPosition.Profit() + g_eaPosition.Swap() + g_eaPosition.Commission();
                    }
                }
            }
            
            status = StringFormat("IN TRADE: $%.2f", symbolProfit);
            textColor = symbolProfit >= 0 ? clrLime : clrRed;
        }
        else if(g_symbolLocks[i].isLocked)
        {
            // Locked (no position but breakout already detected today)
            string lockDate = TimeToString(g_symbolLocks[i].lockedD1Time, TIME_DATE);
            status = StringFormat("? DONE TODAY (%s)", lockDate);
            textColor = clrOrange;
        }
        else
        {
            // Ready to trade
            status = "? MONITORING";
            textColor = clrGray;
        }
        
        string winRate = g_stats[i].totalTrades > 0 ? 
                        StringFormat("%.0f%%", (double)g_stats[i].winningTrades / g_stats[i].totalTrades * 100) : 
                        "-";
        
        string text = StringFormat("%s: %s | W/L: %d/%d (%s)", 
                                  symbol, status,
                                  g_stats[i].winningTrades, 
                                  g_stats[i].losingTrades,
                                  winRate);
        
        ObjectSetString(0, labelName, OBJPROP_TEXT, text);
        ObjectSetInteger(0, labelName, OBJPROP_COLOR, textColor);
    }
    
    ChartRedraw();
}

//| Get Deinitialization Reason Text                                 |
//+------------------------------------------------------------------+
string GetDeinitReasonText(const int reason)
{
    switch(reason)
    {
        case REASON_PROGRAM: return "Program stopped";
        case REASON_REMOVE: return "EA removed from chart";
        case REASON_RECOMPILE: return "EA recompiled";
        case REASON_CHARTCHANGE: return "Chart symbol/period changed";
        case REASON_CHARTCLOSE: return "Chart closed";
        case REASON_PARAMETERS: return "Input parameters changed";
        case REASON_ACCOUNT: return "Account changed";
        case REASON_TEMPLATE: return "Template applied";
        case REASON_INITFAILED: return "Initialization failed";
        case REASON_CLOSE: return "Terminal closing";
        default: return "Unknown reason";
    }
}

//+------------------------------------------------------------------+
//| OnTick - Main loop                                                |
//+------------------------------------------------------------------+
void OnTick()
{
    if(!g_initialized)
        return;
    
    datetime currentTime = TimeCurrent();
    if(currentTime - g_lastCheck < 60)
        return;
    
    g_lastCheck = currentTime;
    
    // Process each symbol
    for(int i = 0; i < SYMBOL_COUNT; i++)
    {
        if(!g_tradingSymbols[i].isActive)
            continue;
        
        string symbol = g_tradingSymbols[i].symbol;
        
        // Check for new D1 candle (auto-unlocks if changed)
        CheckNewHTFCandle(symbol, i);
        
        // Process symbol (respects lock state)
        ProcessSymbol(symbol, i);
    }
    
    if(ShowDashboard)
    {
        UpdateDashboard();
    }
}


//+------------------------------------------------------------------+
//| Count Breakout Positions for Symbol                              |
//+------------------------------------------------------------------+
int CountBreakoutPositions(const string symbol)
{
    int count = 0;
    
    for(int i = 0; i < PositionsTotal(); i++)
    {
        if(g_eaPosition.SelectByIndex(i))
        {
            if(g_eaPosition.Symbol() == symbol && 
               g_eaPosition.Magic() == MagicNumber)
            {
                count++;
            }
        }
    }
    
    return count;
}

//+------------------------------------------------------------------+
//| Main Processing with Persistent Lock                             |
//+------------------------------------------------------------------+
void ProcessSymbol(const string symbol, const int index)
{
    // ================================================================
    // ? GATE #0: Check persistent lock FIRST
    // ================================================================
    if(!g_symbolLocks[index].CanTrade())
    {
        // Symbol is locked - don't process anything
        // This prevents RE-EVALUATION of breakout signals
        return;
    }
    
    // ================================================================
    // GATE #1: Session filter
    // ================================================================
    if(UseSessionFilter && !IsInTradingSession())
        return;
    
    // ================================================================
    // GATE #2: Manage existing positions
    // ================================================================
    ManagePositions(symbol, index);
    
    // ================================================================
    // GATE #3: Position limits
    // ================================================================
    int currentPositions = CountBreakoutPositions(symbol);
    if(currentPositions >= MaxOpenPositions)
        return;
    
    int totalPositions = GetAllPositionsCount();
    if(totalPositions >= MaxTotalPositions)
        return;
    
    // ================================================================
    // GATE #4: Check for breakout
    // ================================================================
    ENUM_ORDER_TYPE orderType;
    double breakLevel;
    
    if(!CheckBreakout(symbol, index, orderType, breakLevel))
    {
        // No breakout detected yet
        return;
    }
    
    // ================================================================
    // ? CRITICAL: LOCK IMMEDIATELY after detecting breakout
    //    This prevents checking for another breakout today
    // ================================================================
    g_symbolLocks[index].Lock();
    
    PrintFormat("? %s: Breakout detected on %s", 
               symbol, 
               TimeToString(iTime(symbol, PERIOD_D1, 0), TIME_DATE));
    PrintFormat("   Symbol locked, now waiting for entry confirmation...");
    
    g_htfData[index].breakLevel = breakLevel;
    
    // ================================================================
    // GATE #5: Wait for entry confirmation
    // ================================================================
    if(!CheckEntryConfirmation(symbol, index, orderType))
    {
        PrintFormat("? %s: No confirmation yet, will retry on next tick", symbol);
        PrintFormat("   (Symbol stays locked - no new breakout checks)");
        return;
    }
    
    // ================================================================
    // GATE #6: Spread filter
    // ================================================================
    if(UseSpreadFilter && !IsSpreadAcceptable(symbol))
    {
        PrintFormat("?? %s: Spread too high, waiting...", symbol);
        PrintFormat("   (Symbol stays locked)");
        return;
    }
    
    // ================================================================
    // GATE #7: Execute trade
    // ================================================================
    ExecuteTrade(symbol, index, orderType, breakLevel);
}

//+------------------------------------------------------------------+
//| ? NEW: Reset daily tracker on new D1 candle                    |
//+------------------------------------------------------------------+
void CheckNewHTFCandle(const string symbol, const int index)
{
    datetime currentHTF = iTime(symbol, PERIOD_D1, 0);
    
    // Check if D1 changed
    if(currentHTF != g_htfData[index].time)
    {
        if(g_htfData[index].time > 0)
        {
            PrintFormat("? %s: NEW D1 CANDLE - Resetting daily tracker", symbol);
            PrintFormat("   Old: %s", TimeToString(g_htfData[index].time, TIME_DATE));
            PrintFormat("   New: %s", TimeToString(currentHTF, TIME_DATE));
            
            // ? CRITICAL: Reset the daily tracker for new day
            g_dailyTracker[index].Reset();
        }
        
        if(!MarkPreviousHTF(symbol, index))
        {
            PrintFormat("?? %s: Failed to mark HTF", symbol);
            return;
        }
        
        PrintFormat("? %s: HTF marked | Tracker reset | Ready for new trade", symbol);
    }
}

//+------------------------------------------------------------------+
//| SIMPLIFIED MarkPreviousHTF - Just update levels, no flags        |
//+------------------------------------------------------------------+
bool MarkPreviousHTF(const string symbol, const int index)
{
    double open[], close[], high[], low[];
    datetime time[];
    
    if(CopyOpen(symbol, PERIOD_D1, 1, 1, open) <= 0 ||
       CopyClose(symbol, PERIOD_D1, 1, 1, close) <= 0 ||
       CopyHigh(symbol, PERIOD_D1, 1, 1, high) <= 0 ||
       CopyLow(symbol, PERIOD_D1, 1, 1, low) <= 0 ||
       CopyTime(symbol, PERIOD_D1, 0, 1, time) <= 0)
    {
        return false;
    }
    
    datetime newD1Time = time[0];
    
    // Log if D1 changed
    if(g_htfData[index].time > 0 && g_htfData[index].time != newD1Time)
    {
        PrintFormat("ðŸ“… %s: NEW D1 CANDLE", symbol);
        PrintFormat("   Previous: %s", TimeToString(g_htfData[index].time, TIME_DATE));
        PrintFormat("   Current:  %s", TimeToString(newD1Time, TIME_DATE));
    }
    
    // Simply update HTF data - no flag resets needed
    g_htfData[index].open = open[0];
    g_htfData[index].close = close[0];
    g_htfData[index].high = high[0];
    g_htfData[index].low = low[0];
    g_htfData[index].time = newD1Time;
    
    if(symbol == _Symbol)
    {
        DrawHTFLevels(symbol, index);
    }
    
    return true;
}

        return false;
    
    double prevH4Close = closeH4[0];

    if(currentPriceD1 > g_htfData[index].high && 
       prevH4Close > g_htfData[index].high)
    {
        orderType = ORDER_TYPE_BUY;
        breakLevel = g_htfData[index].high;
        
        Print("¨X¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨[");
        PrintFormat("¨U  ? %s: BREAKOUT UP", symbol);
        PrintFormat("¨U     D1 Close: %.5f", currentPriceD1);
        PrintFormat("¨U     D1 High:  %.5f", g_htfData[index].high);
        PrintFormat("¨U     Diff:     %.5f", currentPriceD1 - g_htfData[index].high);
        Print("¨^¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨a");
        return true;
    }
    
    // Check BREAKOUT DOWN
    if(currentPriceD1 < g_htfData[index].low && 
       prevH4Close < g_htfData[index].low)
    {
        orderType = ORDER_TYPE_SELL;
        breakLevel = g_htfData[index].low;

        PrintFormat("¨U     Diff:     %.5f", g_htfData[index].low - currentPriceD1);

//+------------------------------------------------------------------+
//| Check Entry Confirmation                                          |
//+------------------------------------------------------------------+
bool CheckEntryConfirmation(const string symbol, const int index, const ENUM_ORDER_TYPE orderType)
{
    double keyLevel = g_htfData[index].breakLevel;
    
    RejectionSignal signal = PricePosition(symbol, keyLevel);
    
    if(!CheckADXFilter(symbol, index))
    {
        return false;
    }
    
    // For BUY, need bullish rejection
    if(orderType == ORDER_TYPE_BUY && signal.isBullishRejection)
    {
        PrintFormat("%s: BUY confirmation - %s rejection + ADX passed", 
                   symbol, signal.rejectionType);
        return true;
    }
    
    // For SELL, need bearish rejection
    if(orderType == ORDER_TYPE_SELL && signal.isBearishRejection)
    {
        PrintFormat("%s: SELL confirmation - %s rejection + ADX passed", 
                   symbol, signal.rejectionType);
        return true;
    }
    
    return false;
}

//+------------------------------------------------------------------+
//| Execute Trade - Lock already set                                 |
//+------------------------------------------------------------------+
bool ExecuteTrade(const string symbol, const int index, const ENUM_ORDER_TYPE orderType, const double breakLevel)
{
    if(!g_eaSymbolInfo.Name(symbol))
        return false;
    
    g_eaSymbolInfo.RefreshRates();
    
    double atr = GetATRValue(index);
    if(atr <= 0)
    {
        PrintFormat("? %s: Invalid ATR, cannot calculate position size", symbol);
        return false;
    }
    
    double lotSize = CalculateLotSize(symbol, index, atr);
    if(lotSize < g_eaSymbolInfo.LotsMin())
    {
        PrintFormat("? %s: Lot size too small: %.2f", symbol, lotSize);
        return false;
    }
    
    double price, sl, tp;
    string direction;
    
    if(orderType == ORDER_TYPE_BUY)
    {
        direction = "BUY";
        price = g_eaSymbolInfo.Ask();
        sl = NormalizeDouble(price - (atr * ATRStopLoss), g_eaSymbolInfo.Digits());
        tp = NormalizeDouble(price + (atr * ATRTakeProfit), g_eaSymbolInfo.Digits());
    }
    else
    {
        direction = "SELL";
        price = g_eaSymbolInfo.Bid();
        sl = NormalizeDouble(price + (atr * ATRStopLoss), g_eaSymbolInfo.Digits());
        tp = NormalizeDouble(price - (atr * ATRTakeProfit), g_eaSymbolInfo.Digits());
    }
    
    datetime currentD1 = iTime(symbol, PERIOD_D1, 0);
    string dateStr = TimeToString(currentD1, TIME_DATE);
    string tradeComment = StringFormat("BO_%s_%s", direction, dateStr);
    
    bool result = g_eaTrade.PositionOpen(symbol, orderType, lotSize, price, sl, tp, tradeComment);
    
    if(result && g_eaTrade.ResultRetcode() == TRADE_RETCODE_DONE)
    {
        ulong ticket = g_eaTrade.ResultOrder();
        double riskPips = MathAbs(price - sl) / g_eaSymbolInfo.Point() / 10;
        double rewardPips = MathAbs(tp - price) / g_eaSymbolInfo.Point() / 10;
        
        Print("¨X¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨[");
        PrintFormat("¨U  ? TRADE EXECUTED: %s %s", symbol, direction);
        PrintFormat("¨U     Ticket: #%I64u", ticket);
        PrintFormat("¨U     Date: %s", dateStr);
        PrintFormat("¨U     Lot Size: %.2f", lotSize);
        PrintFormat("¨U     Entry Price: %.5f", price);
        PrintFormat("¨U     Stop Loss: %.5f (%.1f pips)", sl, riskPips);
        PrintFormat("¨U     Take Profit: %.5f (%.1f pips)", tp, rewardPips);
        PrintFormat("¨U     Risk/Reward: 1:%.1f", rewardPips/riskPips);
        PrintFormat("¨U     Comment: %s", tradeComment);
        PrintFormat("¨U");
        PrintFormat("¨U     ? Symbol locked until next D1 candle");
        PrintFormat("¨U     ? Lock saved to persistent storage");
        Print("¨^¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨T¨a");
        
        g_stats[index].totalTrades++;
        g_stats[index].lastTradeTime = TimeCurrent();
        
        return true;
    }
    else
    {
        PrintFormat("? TRADE FAILED: %s %s", symbol, direction);
        PrintFormat("   Retcode: %d (%s)", 
                   g_eaTrade.ResultRetcode(), 
                   g_eaTrade.ResultRetcodeDescription());
        PrintFormat("   Note: Symbol stays LOCKED (breakout already detected)");
        return false;
    }
}
