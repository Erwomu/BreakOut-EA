//--- Input Parameters
input group "=== Core Settings ==="
input int      InpMagicNumber = 234567;           // Magic Number
input double   InpRiskPercent = 1.0;              // Risk per trade (%)
input int      InpMaxTradesPerDay = 1;            // Max trades per day
input bool     InpVerboseLog = true;              // Verbose logging

input group "=== ATR Settings ==="
input int      InpATRPeriod = 14;                 // ATR Period (H1)
input double   InpATRStopLoss = 2.0;              // ATR Stop Loss Multiplier
input double   InpATRTakeProfit = 3.0;            // ATR Take Profit Multiplier

input group "=== Trailing Stop ==="
input double   InpBEMultiplier = 1.0;             // Breakeven ATR Multiplier
input double   InpTrailStartMultiplier = 1.5;     // Trail Start ATR Multiplier
input double   InpTrailStopMultiplier = 0.5;      // Trail Distance ATR Multiplier

input group "=== ADX Filter ==="
input bool     InpUseADXFilter = false;           // Use ADX Filter (OFF for testing)
input int      InpADXPeriod = 14;                 // ADX Period (H4)
input double   InpMinADX = 20.0;                  // Minimum ADX Value (lowered)

input group "=== Session Filter ==="
input bool     InpUseSessionFilter = false;       // Use Session Filter (OFF for testing)
input int      InpLondonStart = 8;                // London Start Hour
input int      InpLondonEnd = 17;                 // London End Hour
input int      InpNewYorkStart = 13;              // New York Start Hour
input int      InpNewYorkEnd = 22;                // New York End Hour

input group "=== Spread Filter ==="
input bool     InpUseSpreadFilter = false;        // Use Spread Filter (OFF for testing)
input double   InpMaxSpreadPips = 10.0;           // Max Spread (pips)

//--- Trading State Structure
struct TradingState {
    datetime d1Time;                    // Current D1 candle time
    double previousD1High;              // Previous D1 high level
    double previousD1Low;               // Previous D1 low level
    
    datetime lastCheckedH4Time;         // Last H4 candle we checked for breakout
    
    // Pending entry state
    bool waitingForH1Entry;             // Are we waiting for H1 entry?
    datetime targetH1Time;              // Target H1 candle time for entry
    ENUM_ORDER_TYPE pendingDirection;   // Direction of pending trade
    double pendingBreakLevel;           // Break level for logging
    
    void ResetDaily() {
        lastCheckedH4Time = 0;
        waitingForH1Entry = false;
        targetH1Time = 0;
    }
    
    void ResetAll() {
        d1Time = 0;
        previousD1High = 0;
        previousD1Low = 0;
        ResetDaily();
    }
};

//--- Global Variables
TradingState g_state;
CTrade g_trade;
CSymbolInfo g_symbol;
CPositionInfo g_position;

int g_atrHandle = INVALID_HANDLE;
int g_adxHandle = INVALID_HANDLE;

double g_atrBuffer[];
double g_adxBuffer[];

int g_tradesToday = 0;
datetime g_lastTradeDate = 0;

const string GV_TRADE_COUNT = "BO_TradeCount";
const string GV_TRADE_DATE = "BO_TradeDate";

int g_h4CheckCount = 0;
int g_breakoutCount = 0;

//+------------------------------------------------------------------+
//| Expert initialization function                                   |
//+------------------------------------------------------------------+
int OnInit()
{
    // Initialize symbol info
    if(!g_symbol.Name(_Symbol)) {
        Print("❌ ERROR: Failed to initialize symbol info");
        return INIT_FAILED;
    }
    
    // Set trade parameters
    g_trade.SetExpertMagicNumber(InpMagicNumber);
    g_trade.SetDeviationInPoints(10);
    g_trade.SetTypeFilling(ORDER_FILLING_IOC);  // Changed to IOC (more compatible)
    g_trade.SetAsyncMode(false);
    
    // Initialize ATR indicator (H1 timeframe)
    g_atrHandle = iATR(_Symbol, PERIOD_H1, InpATRPeriod);
    if(g_atrHandle == INVALID_HANDLE) {
        Print("❌ ERROR: Failed to create ATR indicator");
        return INIT_FAILED;
    }
    
    // Initialize ADX indicator (H4 timeframe)
    if(InpUseADXFilter) {
        g_adxHandle = iADX(_Symbol, PERIOD_H4, InpADXPeriod);
        if(g_adxHandle == INVALID_HANDLE) {
            Print("❌ ERROR: Failed to create ADX indicator");
            return INIT_FAILED;
        }
    }
    
    // Set buffer as series
    ArraySetAsSeries(g_atrBuffer, true);
    ArraySetAsSeries(g_adxBuffer, true);
    
    // Initialize state
    g_state.ResetAll();
    
    // Recover daily trade count
    RecoverDailyTradeCount();
    
    Print("✅ EA Initialized Successfully");
    Print("📊 Symbol: ", _Symbol);
    Print("🎯 Magic Number: ", InpMagicNumber);
    Print("💰 Risk per trade: ", InpRiskPercent, "%");
    Print("📅 Max trades per day: ", InpMaxTradesPerDay);
    Print("🔍 ADX Filter: ", InpUseADXFilter ? "ON" : "OFF");
    Print("⏰ Session Filter: ", InpUseSessionFilter ? "ON" : "OFF");
    Print("📶 Spread Filter: ", InpUseSpreadFilter ? "ON" : "OFF");
    Print("📝 Verbose Logging: ", InpVerboseLog ? "ON" : "OFF");
    
    return INIT_SUCCEEDED;
}

//+------------------------------------------------------------------+
//| Expert deinitialization function                                 |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
    // Release indicator handles
    if(g_atrHandle != INVALID_HANDLE)
        IndicatorRelease(g_atrHandle);
    if(g_adxHandle != INVALID_HANDLE)
        IndicatorRelease(g_adxHandle);
    
    Print("👋 EA Deinitialized. Reason: ", reason);
    Print("📊 Statistics:");
    Print("   H4 candles checked: ", g_h4CheckCount);
    Print("   Breakouts detected: ", g_breakoutCount);
    Print("   Trades executed: ", g_tradesToday);
}

//+------------------------------------------------------------------+
//| Expert tick function                                             |
//+------------------------------------------------------------------+
void OnTick()
{
    // Step 1: Check for new D1 candle and update levels
    CheckNewD1Candle();
    
    // Step 2: Check if daily limit allows trading
    if(!CanTradeToday()) {
        return;
    }
    
    // Step 3: If waiting for H1 entry, check if it's time
    if(g_state.waitingForH1Entry) {
        CheckPendingEntry();
        return; // Don't check for new breakouts while waiting
    }
    
    // Step 4: Check for H4 breakout
    CheckForBreakout();
    
    // Step 5: Manage open positions
    ManagePositions();
}

//+------------------------------------------------------------------+
//| Check for new D1 candle and mark previous levels                |
//+------------------------------------------------------------------+
void CheckNewD1Candle()
{
    datetime d1Time = iTime(_Symbol, PERIOD_D1, 0);
    
    // Check if we have a new D1 candle
    if(d1Time != g_state.d1Time) {
        // Get previous D1 high and low
        double prevHigh = iHigh(_Symbol, PERIOD_D1, 1);
        double prevLow = iLow(_Symbol, PERIOD_D1, 1);
        
        if(prevHigh > 0 && prevLow > 0) {
            g_state.d1Time = d1Time;
            g_state.previousD1High = prevHigh;
            g_state.previousD1Low = prevLow;
            g_state.ResetDaily();
            
            Print("📅 NEW DAY: ", TimeToString(d1Time, TIME_DATE));
            Print("   Previous D1 High: ", DoubleToString(prevHigh, _Digits));
            Print("   Previous D1 Low: ", DoubleToString(prevLow, _Digits));
            Print("   Range: ", DoubleToString((prevHigh - prevLow) / _Point, 0), " points");
            Print("   Ready to monitor for breakouts");
        }
    }
}

//+------------------------------------------------------------------+
//| Check for H4 breakout                                            |
//+------------------------------------------------------------------+
void CheckForBreakout()
{
    // Need valid D1 levels first
    if(g_state.previousD1High == 0 || g_state.previousD1Low == 0) {
        if(InpVerboseLog && TimeCurrent() - g_state.d1Time < 3600) {
            Print("⏳ Waiting for valid D1 levels...");
        }
        return;
    }
    
    // Get last CLOSED H4 candle
    datetime h4Time = iTime(_Symbol, PERIOD_H4, 1);
    
    // Skip if we already checked this H4 candle
    if(h4Time <= g_state.lastCheckedH4Time)
        return;
    
    double h4Close = iClose(_Symbol, PERIOD_H4, 1);
    if(h4Close == 0)
        return;
    
    g_h4CheckCount++;
    
    if(InpVerboseLog) {
        Print("🔍 Checking H4 candle: ", TimeToString(h4Time, TIME_DATE|TIME_MINUTES));
        Print("   H4 Close: ", DoubleToString(h4Close, _Digits));
        Print("   D1 High: ", DoubleToString(g_state.previousD1High, _Digits), 
              " (", DoubleToString((g_state.previousD1High - h4Close) / _Point, 0), " points away)");
        Print("   D1 Low: ", DoubleToString(g_state.previousD1Low, _Digits),
              " (", DoubleToString((h4Close - g_state.previousD1Low) / _Point, 0), " points away)");
    }
    
    bool breakoutDetected = false;
    
    // Check for bullish breakout
    if(h4Close > g_state.previousD1High) {
        g_state.pendingDirection = ORDER_TYPE_BUY;
        g_state.pendingBreakLevel = g_state.previousD1High;
        breakoutDetected = true;
        g_breakoutCount++;
        
        Print("🔥 H4 BREAKOUT UP detected!");
        Print("   H4 Close: ", DoubleToString(h4Close, _Digits));
        Print("   Break Level: ", DoubleToString(g_state.pendingBreakLevel, _Digits));
        Print("   Breakout by: ", DoubleToString((h4Close - g_state.pendingBreakLevel) / _Point, 0), " points");
    }
    // Check for bearish breakout
    else if(h4Close < g_state.previousD1Low) {
        g_state.pendingDirection = ORDER_TYPE_SELL;
        g_state.pendingBreakLevel = g_state.previousD1Low;
        breakoutDetected = true;
        g_breakoutCount++;
        
        Print("🔥 H4 BREAKOUT DOWN detected!");
        Print("   H4 Close: ", DoubleToString(h4Close, _Digits));
        Print("   Break Level: ", DoubleToString(g_state.pendingBreakLevel, _Digits));
        Print("   Breakout by: ", DoubleToString((g_state.pendingBreakLevel - h4Close) / _Point, 0), " points");
    }
    
    // Update last checked time
    g_state.lastCheckedH4Time = h4Time;
    
    // If breakout detected, schedule entry for next H1
    if(breakoutDetected) {
        datetime currentH1Time = iTime(_Symbol, PERIOD_H1, 0);
        g_state.targetH1Time = currentH1Time + PeriodSeconds(PERIOD_H1);
        g_state.waitingForH1Entry = true;
        
        Print("⏰ Scheduled entry for H1 candle at: ", TimeToString(g_state.targetH1Time, TIME_DATE|TIME_MINUTES));
    }
}

//+------------------------------------------------------------------+
//| Check if it's time to execute pending entry                      |
//+------------------------------------------------------------------+
void CheckPendingEntry()
{
    datetime currentH1Time = iTime(_Symbol, PERIOD_H1, 0);
    
    // Check if we've reached or passed the target H1 time
    if(currentH1Time >= g_state.targetH1Time) {
        Print("✅ Target H1 candle reached - Executing trade now");
        ExecuteTrade(g_state.pendingDirection, g_state.pendingBreakLevel);
        g_state.waitingForH1Entry = false; // Reset waiting state
    }
}

//+------------------------------------------------------------------+
//| Execute trade                                                     |
//+------------------------------------------------------------------+
bool ExecuteTrade(ENUM_ORDER_TYPE direction, double breakLevel)
{
    Print("📋 Attempting to execute trade...");
    
    // Final check: Can we trade?
    if(!CanTradeToday()) {
        Print("🛑 DAILY LIMIT REACHED: ", g_tradesToday, "/", InpMaxTradesPerDay, " trades");
        return false;
    }
    
    // Check filters
    if(!CheckFilters()) {
        Print("❌ Trade filters not passed");
        return false;
    }
    
    // Get ATR for position sizing and SL/TP
    double atr = GetATR();
    if(atr <= 0) {
        Print("❌ ERROR: Invalid ATR value");
        return false;
    }
    
    // Calculate position size
    double lotSize = CalculateLotSize(atr);
    if(lotSize <= 0) {
        Print("❌ ERROR: Invalid lot size calculated");
        return false;
    }
    
    // Get current price
    double price = (direction == ORDER_TYPE_BUY) ? g_symbol.Ask() : g_symbol.Bid();
    
    // Calculate SL and TP
    double sl, tp;
    if(direction == ORDER_TYPE_BUY) {
        sl = price - (atr * InpATRStopLoss);
        tp = price + (atr * InpATRTakeProfit);
    } else {
        sl = price + (atr * InpATRStopLoss);
        tp = price - (atr * InpATRTakeProfit);
    }
    
    // Normalize prices
    sl = NormalizeDouble(sl, _Digits);
    tp = NormalizeDouble(tp, _Digits);
    
    // Create comment
    string comment = StringFormat("BO_%s_%s", 
                                  (direction == ORDER_TYPE_BUY) ? "BUY" : "SELL",
                                  TimeToString(TimeCurrent(), TIME_DATE));
    
    Print("✅ ENTERING TRADE at H1 open");
    Print("   Direction: ", (direction == ORDER_TYPE_BUY) ? "BUY" : "SELL");
    Print("   Lot Size: ", DoubleToString(lotSize, 2));
    Print("   Entry: ", DoubleToString(price, _Digits));
    Print("   SL: ", DoubleToString(sl, _Digits), " (", DoubleToString(MathAbs(price - sl) / _Point, 1), " points)");
    Print("   TP: ", DoubleToString(tp, _Digits), " (", DoubleToString(MathAbs(tp - price) / _Point, 1), " points)");
    Print("   ATR: ", DoubleToString(atr, _Digits));
    Print("   Risk: $", DoubleToString(AccountInfoDouble(ACCOUNT_BALANCE) * InpRiskPercent / 100.0, 2));
    
    // Execute trade
    bool result = false;
    if(direction == ORDER_TYPE_BUY) {
        result = g_trade.Buy(lotSize, _Symbol, price, sl, tp, comment);
    } else {
        result = g_trade.Sell(lotSize, _Symbol, price, sl, tp, comment);
    }
    
    if(result) {
        Print("📊 Trade executed successfully!");
        Print("   Ticket: ", g_trade.ResultOrder());
        IncrementTradeCount();
        return true;
    } else {
        Print("❌ Trade execution failed!");
        Print("   Error: ", GetLastError());
        Print("   Return code: ", g_trade.ResultRetcode());
        Print("   Comment: ", g_trade.ResultRetcodeDescription());
        return false;
    }
}

//+------------------------------------------------------------------+
//| Manage open positions (trailing stop)                            |
//+------------------------------------------------------------------+
void ManagePositions()
{
    for(int i = PositionsTotal() - 1; i >= 0; i--) {
        if(!g_position.SelectByIndex(i))
            continue;
        
        if(g_position.Symbol() != _Symbol || g_position.Magic() != InpMagicNumber)
            continue;
        
        double atr = GetATR();
        if(atr <= 0)
            continue;
        
        double entryPrice = g_position.PriceOpen();
        double currentSL = g_position.StopLoss();
        double currentPrice = (g_position.Type() == POSITION_TYPE_BUY) ? 
                              g_symbol.Bid() : g_symbol.Ask();
        
        double profit = (g_position.Type() == POSITION_TYPE_BUY) ? 
                        (currentPrice - entryPrice) : (entryPrice - currentPrice);
        
        double newSL = currentSL;
        bool updateSL = false;
        
        // Move to breakeven
        if(profit >= atr * InpBEMultiplier) {
            if(g_position.Type() == POSITION_TYPE_BUY) {
                if(currentSL < entryPrice) {
                    newSL = entryPrice;
                    updateSL = true;
                    Print("🔒 Moving to BREAKEVEN for ticket ", g_position.Ticket());
                }
            } else {
                if(currentSL > entryPrice || currentSL == 0) {
                    newSL = entryPrice;
                    updateSL = true;
                    Print("🔒 Moving to BREAKEVEN for ticket ", g_position.Ticket());
                }
            }
        }
        
        // Start trailing
        if(profit >= atr * InpTrailStartMultiplier) {
            if(g_position.Type() == POSITION_TYPE_BUY) {
                double trailSL = currentPrice - (atr * InpTrailStopMultiplier);
                if(trailSL > currentSL && trailSL > entryPrice) {
                    newSL = trailSL;
                    updateSL = true;
                    Print("📈 Trailing stop for ticket ", g_position.Ticket());
                }
            } else {
                double trailSL = currentPrice + (atr * InpTrailStopMultiplier);
                if(trailSL < currentSL || currentSL == 0) {
                    if(trailSL < entryPrice) {
                        newSL = trailSL;
                        updateSL = true;
                        Print("📉 Trailing stop for ticket ", g_position.Ticket());
                    }
                }
            }
        }
        
        // Update SL if needed
        if(updateSL) {
            newSL = NormalizeDouble(newSL, _Digits);
            if(g_trade.PositionModify(g_position.Ticket(), newSL, g_position.TakeProfit())) {
                Print("   New SL: ", DoubleToString(newSL, _Digits));
            } else {
                Print("❌ Failed to modify SL. Error: ", GetLastError());
            }
        }
    }
}

//+------------------------------------------------------------------+
//| Get ATR value                                                     |
//+------------------------------------------------------------------+
double GetATR()
{
    if(CopyBuffer(g_atrHandle, 0, 0, 2, g_atrBuffer) <= 0) {
        Print("❌ ERROR: Failed to copy ATR buffer");
        return 0;
    }
    return g_atrBuffer[0];
}

//+------------------------------------------------------------------+
//| Get ADX value                                                     |
//+------------------------------------------------------------------+
double GetADX()
{
    if(g_adxHandle == INVALID_HANDLE)
        return 0;
    
    if(CopyBuffer(g_adxHandle, 0, 0, 2, g_adxBuffer) <= 0) {
        Print("❌ ERROR: Failed to copy ADX buffer");
        return 0;
    }
    return g_adxBuffer[0];
}

//+------------------------------------------------------------------+
//| Check if we can trade today                                      |
//+------------------------------------------------------------------+
bool CanTradeToday()
{
    datetime todayKey = TodayKey();
    
    // Check if new day
    if(todayKey != g_lastTradeDate) {
        g_tradesToday = 0;
        g_lastTradeDate = todayKey;
        GlobalVariableSet(GV_TRADE_COUNT, 0);
        GlobalVariableSet(GV_TRADE_DATE, (double)todayKey);
    }
    
    return (g_tradesToday < InpMaxTradesPerDay);
}

//+------------------------------------------------------------------+
//| Increment trade count                                            |
//+------------------------------------------------------------------+
void IncrementTradeCount()
{
    g_tradesToday++;
    GlobalVariableSet(GV_TRADE_COUNT, g_tradesToday);
    Print("📊 Trade count updated: ", g_tradesToday, "/", InpMaxTradesPerDay);
}

//+------------------------------------------------------------------+
//| Recover daily trade count from GlobalVariables                   |
//+------------------------------------------------------------------+
void RecoverDailyTradeCount()
{
    datetime todayKey = TodayKey();
    
    if(GlobalVariableCheck(GV_TRADE_DATE)) {
        datetime storedDate = (datetime)GlobalVariableGet(GV_TRADE_DATE);
        
        if(storedDate == todayKey && GlobalVariableCheck(GV_TRADE_COUNT)) {
            g_tradesToday = (int)GlobalVariableGet(GV_TRADE_COUNT);
            g_lastTradeDate = todayKey;
            Print("♻️ Recovered trade count: ", g_tradesToday, "/", InpMaxTradesPerDay);
        } else {
            // New day, reset
            g_tradesToday = 0;
            g_lastTradeDate = todayKey;
            GlobalVariableSet(GV_TRADE_COUNT, 0);
            GlobalVariableSet(GV_TRADE_DATE, (double)todayKey);
            Print("🆕 New trading day started");
        }
    } else {
        // First run
        g_tradesToday = 0;
        g_lastTradeDate = todayKey;
        GlobalVariableSet(GV_TRADE_COUNT, 0);
        GlobalVariableSet(GV_TRADE_DATE, (double)todayKey);
    }
}

//+------------------------------------------------------------------+
//| Get today's date key (midnight)                                  |
//+------------------------------------------------------------------+
datetime TodayKey()
{
    MqlDateTime tm;
    TimeToStruct(TimeCurrent(), tm);
    tm.hour = 0;
    tm.min = 0;
    tm.sec = 0;
    return StructToTime(tm);
}

//+------------------------------------------------------------------+
//| Check all filters                                                |
//+------------------------------------------------------------------+
bool CheckFilters()
{
    Print("🔍 Checking filters...");
    
    // ADX Filter
    if(InpUseADXFilter) {
        double adx = GetADX();
        if(adx < InpMinADX) {
            Print("   ❌ ADX filter failed: ", DoubleToString(adx, 2), " < ", InpMinADX);
            return false;
        }
        Print("   ✓ ADX: ", DoubleToString(adx, 2));
    } else {
        Print("   ⚪ ADX filter: OFF");
    }
    
    // Session Filter
    if(InpUseSessionFilter) {
        MqlDateTime tm;
        TimeToStruct(TimeCurrent(), tm);
        int hour = tm.hour;
        
        bool inSession = false;
        if((hour >= InpLondonStart && hour < InpLondonEnd) ||
           (hour >= InpNewYorkStart && hour < InpNewYorkEnd)) {
            inSession = true;
        }
        
        if(!inSession) {
            Print("   ❌ Session filter failed: Outside trading hours (", hour, ":00)");
            return false;
        }
        Print("   ✓ Session: Active (", hour, ":00)");
    } else {
        Print("   ⚪ Session filter: OFF");
    }
    
    // Spread Filter
    if(InpUseSpreadFilter) {
        double spread = (g_symbol.Ask() - g_symbol.Bid()) / _Point / 10.0; // pips
        if(spread > InpMaxSpreadPips) {
            Print("   ❌ Spread filter failed: ", DoubleToString(spread, 1), " > ", InpMaxSpreadPips);
            return false;
        }
        Print("   ✓ Spread: ", DoubleToString(spread, 1), " pips");
    } else {
        Print("   ⚪ Spread filter: OFF");
    }
    
    Print("✅ All filters passed!");
    return true;
}

//+------------------------------------------------------------------+
//| Calculate lot size based on risk                                 |
//+------------------------------------------------------------------+
double CalculateLotSize(double atr)
{
    double balance = AccountInfoDouble(ACCOUNT_BALANCE);
    double riskAmount = balance * (InpRiskPercent / 100.0);
    
    // Calculate SL distance in account currency
    double slDistance = atr * InpATRStopLoss;
    double tickValue = g_symbol.TickValue();
    double tickSize = g_symbol.TickSize();
    
    // Calculate lot size
    double lotSize = riskAmount / (slDistance / tickSize * tickValue);
    
    // Normalize to symbol requirements
    double minLot = g_symbol.LotsMin();
    double maxLot = g_symbol.LotsMax();
    double lotStep = g_symbol.LotsStep();
    
    lotSize = MathFloor(lotSize / lotStep) * lotStep;
    lotSize = MathMax(minLot, MathMin(maxLot, lotSize));
    
    return lotSize;
}
