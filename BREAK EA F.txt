//--- Input Parameters
input group "=== Core Settings ==="
input int      InpMagicNumber = 234567;           // Magic Number
input double   InpRiskPercent = 1.0;              // Risk per trade (%)
input int      InpMaxTradesPerDay = 1;            // Max trades per day
input bool     InpVerboseLog = true;              // Verbose logging

input group "=== ATR Settings ==="
input int      InpATRPeriod = 14;                 // ATR Period (H1)
input double   InpATRStopLoss = 2.0;              // ATR Stop Loss Multiplier
input double   InpATRTakeProfit = 3.0;            // ATR Take Profit Multiplier
input int      InpMinStopLossPoints = 150;        // Minimum SL distance (points)

input group "=== Trailing Stop ==="
input double   InpBEMultiplier = 1.0;             // Breakeven ATR Multiplier
input double   InpTrailStartMultiplier = 1.5;     // Trail Start ATR Multiplier
input double   InpTrailStopMultiplier = 0.5;      // Trail Distance ATR Multiplier

input group "=== ADX Filter ==="
input bool     InpUseADXFilter = false;           // Use ADX Filter
input int      InpADXPeriod = 14;                 // ADX Period (H4)
input double   InpMinADX = 20.0;                  // Minimum ADX Value

input group "=== Session Filter ==="
input bool     InpUseSessionFilter = false;       // Use Session Filter
input int      InpLondonStart = 8;                // London Start Hour
input int      InpLondonEnd = 17;                 // London End Hour
input int      InpNewYorkStart = 13;              // New York Start Hour
input int      InpNewYorkEnd = 22;                // New York End Hour

input group "=== Spread Filter ==="
input bool     InpUseSpreadFilter = false;        // Use Spread Filter
input double   InpMaxSpreadPips = 10.0;           // Max Spread (pips)

//--- Trading State Structure
struct TradingState {
    datetime d1Time;
    double previousD1High;
    double previousD1Low;
    datetime lastCheckedH4Time;
    
    // CRITICAL: Track if we've already traded today
    datetime lastTradeOpenTime;
    int todayTradeCount;
    
    void ResetDaily() {
        lastCheckedH4Time = 0;
        todayTradeCount = 0;
        lastTradeOpenTime = 0;
    }
    
    void ResetAll() {
        d1Time = 0;
        previousD1High = 0;
        previousD1Low = 0;
        ResetDaily();
    }
};

//--- Global Variables
TradingState g_state;
CTrade g_trade;
CSymbolInfo g_symbol;
CPositionInfo g_position;

int g_atrHandle = INVALID_HANDLE;
int g_adxHandle = INVALID_HANDLE;
double g_atrBuffer[];
double g_adxBuffer[];

const string GV_LAST_TRADE_DATE = "BO_LastTradeDate_";
const string GV_TRADE_COUNT_TODAY = "BO_TradeCountToday_";

int g_h4CheckCount = 0;
int g_breakoutCount = 0;

//+------------------------------------------------------------------+
//| Expert initialization function                                   |
//+------------------------------------------------------------------+
int OnInit()
{
    if(!g_symbol.Name(_Symbol)) {
        Print("❌ ERROR: Failed to initialize symbol info");
        return INIT_FAILED;
    }
    
    g_symbol.Refresh();
    g_symbol.RefreshRates();
    
    g_trade.SetExpertMagicNumber(InpMagicNumber);
    g_trade.SetDeviationInPoints(50);
    g_trade.SetTypeFilling(ORDER_FILLING_FOK);
    g_trade.SetAsyncMode(false);
    g_trade.LogLevel(LOG_LEVEL_ALL);
    
    g_atrHandle = iATR(_Symbol, PERIOD_H1, InpATRPeriod);
    if(g_atrHandle == INVALID_HANDLE) {
        Print("❌ ERROR: Failed to create ATR indicator");
        return INIT_FAILED;
    }
    
    if(InpUseADXFilter) {
        g_adxHandle = iADX(_Symbol, PERIOD_H4, InpADXPeriod);
        if(g_adxHandle == INVALID_HANDLE) {
            Print("❌ ERROR: Failed to create ADX indicator");
            return INIT_FAILED;
        }
    }
    
    ArraySetAsSeries(g_atrBuffer, true);
    ArraySetAsSeries(g_adxBuffer, true);
    
    g_state.ResetAll();
    
    // Recover state from global variables
    RecoverTradingState();
    
    Print("✅ EA Initialized Successfully");
    Print("📊 Symbol: ", _Symbol);
    Print("🎯 Magic Number: ", InpMagicNumber);
    Print("💰 Risk per trade: ", InpRiskPercent, "%");
    Print("📅 Max trades per day: ", InpMaxTradesPerDay);
    Print("📏 Min Stop Distance: ", g_symbol.StopsLevel(), " points");
    Print("💹 Filling Mode: ", GetFillingMode());
    Print("📊 Trades today: ", g_state.todayTradeCount, "/", InpMaxTradesPerDay);
    
    return INIT_SUCCEEDED;
}

//+------------------------------------------------------------------+
//| Expert deinitialization function                                 |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
    if(g_atrHandle != INVALID_HANDLE)
        IndicatorRelease(g_atrHandle);
    if(g_adxHandle != INVALID_HANDLE)
        IndicatorRelease(g_adxHandle);
    
    Print("👋 EA Deinitialized. Reason: ", reason);
    Print("📊 Statistics:");
    Print("   H4 candles checked: ", g_h4CheckCount);
    Print("   Breakouts detected: ", g_breakoutCount);
    Print("   Trades today: ", g_state.todayTradeCount);
}

//+------------------------------------------------------------------+
//| Expert tick function                                             |
//+------------------------------------------------------------------+
void OnTick()
{
    // Always check for new day first
    CheckNewD1Candle();
    
    // Count actual open trades to sync state
    SyncTradeCount();
    
    // Manage existing positions
    ManagePositions();
    
    // Check if we can open new trades today
    if(g_state.todayTradeCount >= InpMaxTradesPerDay) {
        // Don't spam the log
        return;
    }
    
    // Look for breakout opportunities
    CheckForBreakout();
}

//+------------------------------------------------------------------+
//| Check for new D1 candle                                          |
//+------------------------------------------------------------------+
void CheckNewD1Candle()
{
    datetime d1Time = iTime(_Symbol, PERIOD_D1, 0);
    
    if(d1Time != g_state.d1Time) {
        double prevHigh = iHigh(_Symbol, PERIOD_D1, 1);
        double prevLow = iLow(_Symbol, PERIOD_D1, 1);
        
        if(prevHigh > 0 && prevLow > 0) {
            g_state.d1Time = d1Time;
            g_state.previousD1High = prevHigh;
            g_state.previousD1Low = prevLow;
            g_state.ResetDaily();
            
            // Save to global variable
            string gvName = GV_LAST_TRADE_DATE + IntegerToString(InpMagicNumber);
            GlobalVariableSet(gvName, (double)GetTodayDate());
            
            gvName = GV_TRADE_COUNT_TODAY + IntegerToString(InpMagicNumber);
            GlobalVariableSet(gvName, 0);
            
            Print("📅 NEW DAY: ", TimeToString(d1Time, TIME_DATE));
            Print("   Previous D1 High: ", DoubleToString(prevHigh, _Digits));
            Print("   Previous D1 Low: ", DoubleToString(prevLow, _Digits));
            Print("   Range: ", DoubleToString((prevHigh - prevLow) / _Point, 0), " points");
            Print("   Trade count reset: 0/", InpMaxTradesPerDay);
        }
    }
}

//+------------------------------------------------------------------+
//| Sync trade count with actual positions                           |
//+------------------------------------------------------------------+
void SyncTradeCount()
{
    datetime today = GetTodayDate();
    int actualCount = 0;
    
    // Count positions opened today with our magic number
    for(int i = PositionsTotal() - 1; i >= 0; i--) {
        if(!g_position.SelectByIndex(i))
            continue;
        
        if(g_position.Symbol() != _Symbol || g_position.Magic() != InpMagicNumber)
            continue;
        
        datetime openTime = (datetime)g_position.Time();
        datetime openDate = GetDateFromTime(openTime);
        
        if(openDate == today) {
            actualCount++;
        }
    }
    
    // Also check history for closed trades today
    HistorySelect(today, TimeCurrent());
    
    for(int i = HistoryDealsTotal() - 1; i >= 0; i--) {
        ulong ticket = HistoryDealGetTicket(i);
        if(ticket == 0)
            continue;
        
        if(HistoryDealGetInteger(ticket, DEAL_MAGIC) != InpMagicNumber)
            continue;
        
        if(HistoryDealGetString(ticket, DEAL_SYMBOL) != _Symbol)
            continue;
        
        if(HistoryDealGetInteger(ticket, DEAL_ENTRY) == DEAL_ENTRY_IN) {
            datetime dealTime = (datetime)HistoryDealGetInteger(ticket, DEAL_TIME);
            datetime dealDate = GetDateFromTime(dealTime);
            
            if(dealDate == today) {
                actualCount++;
                break; // Only count once per day
            }
        }
    }
    
    // Update our state if it's out of sync
    if(actualCount != g_state.todayTradeCount) {
        g_state.todayTradeCount = actualCount;
        
        string gvName = GV_TRADE_COUNT_TODAY + IntegerToString(InpMagicNumber);
        GlobalVariableSet(gvName, actualCount);
        
        if(InpVerboseLog) {
            Print("🔄 Trade count synced: ", actualCount, "/", InpMaxTradesPerDay);
        }
    }
}

//+------------------------------------------------------------------+
//| Check for H4 breakout                                            |
//+------------------------------------------------------------------+
void CheckForBreakout()
{
    if(g_state.previousD1High == 0 || g_state.previousD1Low == 0)
        return;
    
    datetime h4Time = iTime(_Symbol, PERIOD_H4, 1);
    
    if(h4Time <= g_state.lastCheckedH4Time)
        return;
    
    double h4Close = iClose(_Symbol, PERIOD_H4, 1);
    if(h4Close == 0)
        return;
    
    g_h4CheckCount++;
    g_state.lastCheckedH4Time = h4Time;
    
    if(InpVerboseLog) {
        Print("🔍 Checking H4 candle: ", TimeToString(h4Time, TIME_DATE|TIME_MINUTES));
        Print("   Trades today: ", g_state.todayTradeCount, "/", InpMaxTradesPerDay);
    }
    
    bool breakoutDetected = false;
    ENUM_ORDER_TYPE direction;
    
    if(h4Close > g_state.previousD1High) {
        direction = ORDER_TYPE_BUY;
        breakoutDetected = true;
        g_breakoutCount++;
        
        Print("🔥 H4 BREAKOUT UP detected!");
        Print("   Breakout by: ", DoubleToString((h4Close - g_state.previousD1High) / _Point, 0), " points");
    }
    else if(h4Close < g_state.previousD1Low) {
        direction = ORDER_TYPE_SELL;
        breakoutDetected = true;
        g_breakoutCount++;
        
        Print("🔥 H4 BREAKOUT DOWN detected!");
        Print("   Breakout by: ", DoubleToString((g_state.previousD1Low - h4Close) / _Point, 0), " points");
    }
    
    if(breakoutDetected) {
        // Wait for next H1 candle to open
        datetime currentH1Time = iTime(_Symbol, PERIOD_H1, 0);
        datetime targetH1Time = currentH1Time + PeriodSeconds(PERIOD_H1);
        
        Print("⏰ Waiting for H1 candle at: ", TimeToString(targetH1Time, TIME_DATE|TIME_MINUTES));
        
        // Sleep until next H1 candle (only in strategy tester or non-visual mode)
        if(MQLInfoInteger(MQL_TESTER)) {
            while(iTime(_Symbol, PERIOD_H1, 0) < targetH1Time && !IsStopped()) {
                Sleep(100);
            }
        }
        
        // Execute immediately when H1 opens
        datetime newH1Time = iTime(_Symbol, PERIOD_H1, 0);
        if(newH1Time >= targetH1Time) {
            Print("✅ H1 candle opened - Executing trade");
            ExecuteTrade(direction);
        }
    }
}

//+------------------------------------------------------------------+
//| Execute trade with proper validation                             |
//+------------------------------------------------------------------+
bool ExecuteTrade(ENUM_ORDER_TYPE direction)
{
    Print("📋 Attempting to execute trade...");
    
    // Final check before execution
    SyncTradeCount();
    
    if(g_state.todayTradeCount >= InpMaxTradesPerDay) {
        Print("🛑 CANNOT TRADE: Daily limit reached (", g_state.todayTradeCount, "/", InpMaxTradesPerDay, ")");
        return false;
    }
    
    if(!CheckFilters()) {
        Print("❌ Trade filters not passed");
        return false;
    }
    
    g_symbol.Refresh();
    g_symbol.RefreshRates();
    
    double atr = GetATR();
    if(atr <= 0) {
        Print("❌ ERROR: Invalid ATR value");
        return false;
    }
    
    double lotSize = CalculateLotSize(atr);
    if(lotSize <= 0) {
        Print("❌ ERROR: Invalid lot size");
        return false;
    }
    
    double price = (direction == ORDER_TYPE_BUY) ? g_symbol.Ask() : g_symbol.Bid();
    
    double slDistance = atr * InpATRStopLoss;
    double tpDistance = atr * InpATRTakeProfit;
    
    int stopsLevel = (int)g_symbol.StopsLevel();
    double minStop = stopsLevel * _Point;
    
    double finalSLDistance = MathMax(slDistance, minStop);
    finalSLDistance = MathMax(finalSLDistance, InpMinStopLossPoints * _Point);
    
    double sl, tp;
    if(direction == ORDER_TYPE_BUY) {
        sl = price - finalSLDistance;
        tp = price + (tpDistance * finalSLDistance / slDistance);
    } else {
        sl = price + finalSLDistance;
        tp = price - (tpDistance * finalSLDistance / slDistance);
    }
    
    sl = NormalizeDouble(sl, _Digits);
    tp = NormalizeDouble(tp, _Digits);
    price = NormalizeDouble(price, _Digits);
    
    if(!ValidateStops(direction, price, sl, tp)) {
        Print("❌ Stop loss or take profit validation failed");
        return false;
    }
    
    string comment = StringFormat("BO_%s_%s", 
                                  (direction == ORDER_TYPE_BUY) ? "BUY" : "SELL",
                                  TimeToString(TimeCurrent(), TIME_DATE));
    
    Print("✅ ENTERING TRADE at H1 open");
    Print("   Direction: ", (direction == ORDER_TYPE_BUY) ? "BUY" : "SELL");
    Print("   Lot Size: ", DoubleToString(lotSize, 2));
    Print("   Entry: ", DoubleToString(price, _Digits));
    Print("   SL: ", DoubleToString(sl, _Digits), " (", DoubleToString(MathAbs(price - sl) / _Point, 1), " points)");
    Print("   TP: ", DoubleToString(tp, _Digits), " (", DoubleToString(MathAbs(tp - price) / _Point, 1), " points)");
    
    bool result = false;
    
    // Try different filling modes
    g_trade.SetTypeFilling(ORDER_FILLING_FOK);
    if(direction == ORDER_TYPE_BUY) {
        result = g_trade.Buy(lotSize, _Symbol, price, sl, tp, comment);
    } else {
        result = g_trade.Sell(lotSize, _Symbol, price, sl, tp, comment);
    }
    
    if(!result) {
        g_trade.SetTypeFilling(ORDER_FILLING_IOC);
        if(direction == ORDER_TYPE_BUY) {
            result = g_trade.Buy(lotSize, _Symbol, price, sl, tp, comment);
        } else {
            result = g_trade.Sell(lotSize, _Symbol, price, sl, tp, comment);
        }
    }
    
    if(!result) {
        g_trade.SetTypeFilling(ORDER_FILLING_RETURN);
        if(direction == ORDER_TYPE_BUY) {
            result = g_trade.Buy(lotSize, _Symbol, price, sl, tp, comment);
        } else {
            result = g_trade.Sell(lotSize, _Symbol, price, sl, tp, comment);
        }
    }
    
    if(result) {
        Print("📊 Trade executed successfully!");
        Print("   Ticket: ", g_trade.ResultOrder());
        Print("   Deal: ", g_trade.ResultDeal());
        
        // CRITICAL: Increment count and save state
        g_state.todayTradeCount++;
        g_state.lastTradeOpenTime = TimeCurrent();
        
        string gvName = GV_TRADE_COUNT_TODAY + IntegerToString(InpMagicNumber);
        GlobalVariableSet(gvName, g_state.todayTradeCount);
        
        Print("🛑 DAILY LIMIT REACHED: ", g_state.todayTradeCount, "/", InpMaxTradesPerDay);
        Print("   No more trades will be taken today");
        
        return true;
    } else {
        Print("❌ Trade execution failed!");
        Print("   Error: ", GetLastError());
        Print("   Return code: ", g_trade.ResultRetcode());
        Print("   Comment: ", g_trade.ResultRetcodeDescription());
        return false;
    }
}

//+------------------------------------------------------------------+
//| Validate stop levels                                             |
//+------------------------------------------------------------------+
bool ValidateStops(ENUM_ORDER_TYPE type, double price, double sl, double tp)
{
    double stopsLevel = g_symbol.StopsLevel() * _Point;
    
    if(type == ORDER_TYPE_BUY) {
        if(sl >= price || tp <= price)
            return false;
        if((price - sl) < stopsLevel || (tp - price) < stopsLevel)
            return false;
    } else {
        if(sl <= price || tp >= price)
            return false;
        if((sl - price) < stopsLevel || (price - tp) < stopsLevel)
            return false;
    }
    
    return true;
}

//+------------------------------------------------------------------+
//| Get filling mode name                                            |
//+------------------------------------------------------------------+
string GetFillingMode()
{
    uint filling = g_symbol.TradeFillFlags();
    
    if((filling & SYMBOL_FILLING_FOK) == SYMBOL_FILLING_FOK)
        return "FOK";
    if((filling & SYMBOL_FILLING_IOC) == SYMBOL_FILLING_IOC)
        return "IOC";
    if((filling & SYMBOL_FILLING_RETURN) == SYMBOL_FILLING_RETURN)
        return "RETURN";
    
    return "UNKNOWN";
}

//+------------------------------------------------------------------+
//| Manage positions                                                 |
//+------------------------------------------------------------------+
void ManagePositions()
{
    for(int i = PositionsTotal() - 1; i >= 0; i--) {
        if(!g_position.SelectByIndex(i))
            continue;
        
        if(g_position.Symbol() != _Symbol || g_position.Magic() != InpMagicNumber)
            continue;
        
        double atr = GetATR();
        if(atr <= 0)
            continue;
        
        double entryPrice = g_position.PriceOpen();
        double currentSL = g_position.StopLoss();
        double currentPrice = (g_position.Type() == POSITION_TYPE_BUY) ? 
                              g_symbol.Bid() : g_symbol.Ask();
        
        double profit = (g_position.Type() == POSITION_TYPE_BUY) ? 
                        (currentPrice - entryPrice) : (entryPrice - currentPrice);
        
        double newSL = currentSL;
        bool updateSL = false;
        
        // Move to breakeven
        if(profit >= atr * InpBEMultiplier) {
            if(g_position.Type() == POSITION_TYPE_BUY) {
                if(currentSL < entryPrice) {
                    newSL = entryPrice;
                    updateSL = true;
                }
            } else {
                if(currentSL > entryPrice || currentSL == 0) {
                    newSL = entryPrice;
                    updateSL = true;
                }
            }
        }
        
        // Start trailing
        if(profit >= atr * InpTrailStartMultiplier) {
            if(g_position.Type() == POSITION_TYPE_BUY) {
                double trailSL = currentPrice - (atr * InpTrailStopMultiplier);
                if(trailSL > currentSL && trailSL > entryPrice) {
                    newSL = trailSL;
                    updateSL = true;
                }
            } else {
                double trailSL = currentPrice + (atr * InpTrailStopMultiplier);
                if((trailSL < currentSL || currentSL == 0) && trailSL < entryPrice) {
                    newSL = trailSL;
                    updateSL = true;
                }
            }
        }
        
        if(updateSL) {
            newSL = NormalizeDouble(newSL, _Digits);
            g_trade.PositionModify(g_position.Ticket(), newSL, g_position.TakeProfit());
        }
    }
}

//+------------------------------------------------------------------+
//| Get ATR value                                                     |
//+------------------------------------------------------------------+
double GetATR()
{
    if(CopyBuffer(g_atrHandle, 0, 0, 2, g_atrBuffer) <= 0)
        return 0;
    return g_atrBuffer[0];
}

//+------------------------------------------------------------------+
//| Get ADX value                                                     |
//+------------------------------------------------------------------+
double GetADX()
{
    if(g_adxHandle == INVALID_HANDLE)
        return 0;
    if(CopyBuffer(g_adxHandle, 0, 0, 2, g_adxBuffer) <= 0)
        return 0;
    return g_adxBuffer[0];
}

//+------------------------------------------------------------------+
//| Recover trading state from global variables                      |
//+------------------------------------------------------------------+
void RecoverTradingState()
{
    datetime today = GetTodayDate();
    
    string gvDateName = GV_LAST_TRADE_DATE + IntegerToString(InpMagicNumber);
    string gvCountName = GV_TRADE_COUNT_TODAY + IntegerToString(InpMagicNumber);
    
    if(GlobalVariableCheck(gvDateName)) {
        datetime storedDate = (datetime)GlobalVariableGet(gvDateName);
        
        if(storedDate == today && GlobalVariableCheck(gvCountName)) {
            g_state.todayTradeCount = (int)GlobalVariableGet(gvCountName);
            Print("♻️ Recovered state: ", g_state.todayTradeCount, " trades today");
        } else {
            // New day
            g_state.todayTradeCount = 0;
            GlobalVariableSet(gvDateName, (double)today);
            GlobalVariableSet(gvCountName, 0);
        }
    } else {
        // First run
        g_state.todayTradeCount = 0;
        GlobalVariableSet(gvDateName, (double)today);
        GlobalVariableSet(gvCountName, 0);
    }
}

//+------------------------------------------------------------------+
//| Get today's date (midnight)                                      |
//+------------------------------------------------------------------+
datetime GetTodayDate()
{
    MqlDateTime tm;
    TimeToStruct(TimeCurrent(), tm);
    tm.hour = 0;
    tm.min = 0;
    tm.sec = 0;
    return StructToTime(tm);
}

//+------------------------------------------------------------------+
//| Get date from datetime                                           |
//+------------------------------------------------------------------+
datetime GetDateFromTime(datetime time)
{
    MqlDateTime tm;
    TimeToStruct(time, tm);
    tm.hour = 0;
    tm.min = 0;
    tm.sec = 0;
    return StructToTime(tm);
}

//+------------------------------------------------------------------+
//| Check all filters                                                |
//+------------------------------------------------------------------+
bool CheckFilters()
{
    if(InpVerboseLog)
        Print("🔍 Checking filters...");
    
    if(InpUseADXFilter) {
        double adx = GetADX();
        if(adx < InpMinADX) {
            Print("   ❌ ADX filter failed: ", DoubleToString(adx, 2));
            return false;
        }
        if(InpVerboseLog)
            Print("   ✓ ADX: ", DoubleToString(adx, 2));
    }
    
    if(InpUseSessionFilter) {
        MqlDateTime tm;
        TimeToStruct(TimeCurrent(), tm);
        int hour = tm.hour;
        
        bool inSession = false;
        if((hour >= InpLondonStart && hour < InpLondonEnd) ||
           (hour >= InpNewYorkStart && hour < InpNewYorkEnd)) {
            inSession = true;
        }
        
        if(!inSession) {
            Print("   ❌ Session filter failed");
            return false;
        }
        if(InpVerboseLog)
            Print("   ✓ Session: Active");
    }
    
    if(InpUseSpreadFilter) {
        double spread = (g_symbol.Ask() - g_symbol.Bid()) / _Point / 10.0;
        if(spread > InpMaxSpreadPips) {
            Print("   ❌ Spread filter failed: ", DoubleToString(spread, 1));
            return false;
        }
        if(InpVerboseLog)
            Print("   ✓ Spread: ", DoubleToString(spread, 1), " pips");
    }
    
    if(InpVerboseLog)
        Print("✅ All filters passed!");
    return true;
}

//+------------------------------------------------------------------+
//| Calculate lot size based on risk                                 |
//+------------------------------------------------------------------+
double CalculateLotSize(double atr)
{
    double balance = AccountInfoDouble(ACCOUNT_BALANCE);
    double riskAmount = balance * (InpRiskPercent / 100.0);
    
    double slDistance = atr * InpATRStopLoss;
    double minSlDistance = InpMinStopLossPoints * _Point;
    slDistance = MathMax(slDistance, minSlDistance);
    
    double tickValue = g_symbol.TickValue();
    double tickSize = g_symbol.TickSize();
    
    double lotSize = riskAmount / (slDistance / tickSize * tickValue);
    
    double minLot = g_symbol.LotsMin();
    double maxLot = g_symbol.LotsMax();
    double lotStep = g_symbol.LotsStep();
    
    lotSize = MathFloor(lotSize / lotStep) * lotStep;
    lotSize = MathMax(minLot, MathMin(maxLot, lotSize));
    
    return lotSize;
}
