datetime TodayKey()
{
    MqlDateTime tm;
    TimeToStruct(TimeCurrent(), tm);
    tm.hour = 0;
    tm.min = 0;
    tm.sec = 0;
    return StructToTime(tm);
}

struct PersistentBreakoutLock
{
    string symbol;
    string storageKey;
    datetime lockedD1Time;
    bool isLocked;
    bool isProcessing;
    
    void Init(const string sym)
    {
        symbol = sym;
        storageKey = "BO_Lock_" + sym;
        lockedD1Time = 0;
        isLocked = false;
        isProcessing = false;
        RecoverFromStorage();
    }
    
    void RecoverFromStorage()
    {
        datetime storedTime = (datetime)GlobalVariableGet(storageKey);
        
        if(storedTime > 0)
        {
            datetime today = TodayKey();  // FIXED: Use calendar day
            
            if(storedTime == today)
            {
                lockedD1Time = storedTime;
                isLocked = true;
                
                Print("┌────────────────────────────────────────┐");
                PrintFormat("│  LOCKED %s: LOCK RECOVERED", symbol);
                PrintFormat("│     From: GlobalVariable storage");
                PrintFormat("│     Date: %s", TimeToString(storedTime, TIME_DATE));
                PrintFormat("│     Status: LOCKED (no trading today)");
                Print("└────────────────────────────────────────┘");
            }
            else
            {
                GlobalVariableDel(storageKey);
                PrintFormat("OK %s: Old lock cleared (new day)", symbol);
            }
        }
        else
        {
            PrintFormat("OK %s: No previous lock found (ready to trade)", symbol);
        }
    }
    
    bool CanTrade()
    {
        datetime today = TodayKey();  // FIXED: Use calendar day
        
        // Check if new day (unlock condition)
        if(today != lockedD1Time && lockedD1Time > 0)
        {
            Print("┌────────────────────────────────────────┐");
            PrintFormat("│  CALENDAR %s: NEW DAY", symbol);
            PrintFormat("│     Previous: %s", TimeToString(lockedD1Time, TIME_DATE));
            PrintFormat("│     Current:  %s", TimeToString(today, TIME_DATE));
            PrintFormat("│     Status: UNLOCKED (ready for new trade)");
            Print("└────────────────────────────────────────┘");
            Unlock();
            return true;
        }
        
        // If locked, cannot trade
        if(isLocked)
        {
            static datetime lastLog = 0;
            if(TimeCurrent() - lastLog > 3600)
            {
                PrintFormat("LOCKED %s: Still locked for %s", 
                           symbol, TimeToString(lockedD1Time, TIME_DATE));
                lastLog = TimeCurrent();
            }
            return false;
        }
        
        // If currently processing, cannot trade (prevents concurrent attempts)
        if(isProcessing)
        {
            return false;
        }
        
        // Check storage sync
        datetime storedTime = (datetime)GlobalVariableGet(storageKey);
        if(storedTime == today && storedTime > 0)
        {
            lockedD1Time = storedTime;
            isLocked = true;
            PrintFormat("WARNING %s: Lock state synced from storage", symbol);
            return false;
        }
        
        return true;
    }
    
    void Lock()
    {
        datetime today = TodayKey();  // FIXED: Use calendar day
        lockedD1Time = today;
        isLocked = true;
        isProcessing = false;
        
        if(!GlobalVariableSet(storageKey, (double)today))
        {
            PrintFormat("WARNING: Failed to save lock to storage for %s", symbol);
        }
        
        datetime stored = (datetime)GlobalVariableGet(storageKey);
        bool storageOK = (stored == today);
        
        Print("┌────────────────────────────────────────┐");
        PrintFormat("│  LOCKED %s LOCKED FOR TODAY", symbol);
        PrintFormat("│     Lock Time: %s", TimeToString(today, TIME_DATE|TIME_MINUTES));
        PrintFormat("│     Storage Key: %s", storageKey);
        PrintFormat("│     Storage Status: %s", storageOK ? "OK SAVED" : "REJECT FAILED");
        PrintFormat("│     Memory Status: OK LOCKED");
        PrintFormat("│");
        PrintFormat("│     STOP NO MORE PROCESSING TODAY");
        PrintFormat("│     CLOCK Will unlock at midnight");
        Print("└────────────────────────────────────────┘");
    }
    
    void Unlock()
    {
        isLocked = false;
        lockedD1Time = 0;
        isProcessing = false;
        
        if(GlobalVariableCheck(storageKey))
        {
            GlobalVariableDel(storageKey);
        }
        
        PrintFormat("UNLOCK %s: UNLOCKED (ready for new trade)", symbol);
    }
    
    void SetProcessing(bool flag)
    {
        isProcessing = flag;
    }
    
    string GetStatus()
    {
        if(isLocked)
        {
            return StringFormat("LOCKED (%s)", TimeToString(lockedD1Time, TIME_DATE));
        }
        if(isProcessing)
        {
            return "WAIT PROCESSING";
        }
        return "OK READY";
    }
};

//| Price Position Detection                                          |
//+------------------------------------------------------------------+
RejectionSignal PricePosition(const string symbol, double keyLevel)
{
    RejectionSignal signal;
    signal.Reset();
    
    double open[], close[], high[], low[];
    datetime time[];
    
    if(CopyOpen(symbol, PERIOD_H1, 1, 3, open) < 3 ||
       CopyClose(symbol, PERIOD_H1, 1, 3, close) < 3 ||
       CopyHigh(symbol, PERIOD_H1, 1, 3, high) < 3 ||
       CopyLow(symbol, PERIOD_H1, 1, 3, low) < 3 ||
       CopyTime(symbol, PERIOD_H1, 1, 3, time) < 3)
    {
        Print("ERROR: Failed to copy price data for ", symbol);
        return signal;
    }
    
    double o1 = open[0],  c1 = close[0],  h1 = high[0],  l1 = low[0];
    double o2 = open[1],  c2 = close[1],  h2 = high[1],  l2 = low[1];
    
    signal.signalTime = time[0];
    
    double body1 = MathAbs(c1 - o1);
    double upperWick1 = h1 - MathMax(o1, c1);
    double lowerWick1 = MathMin(o1, c1) - l1;
    double totalRange1 = h1 - l1;
    
    if(totalRange1 < 0.00001)
        return signal;
    
    double bodyRatio = body1 / totalRange1;
    double upperWickRatio = upperWick1 / totalRange1;
    double lowerWickRatio = lowerWick1 / totalRange1;
    
    if(bodyRatio < 0.25)
        return signal;
    
    if(lowerWickRatio > 0.5 && bodyRatio < 0.4)
    {
        signal.isBullishRejection = true;
        signal.rejectionType = "PIN_BAR";
    }
    else if(c1 > o1 && o2 > c2 && c1 > o2 && o1 >= c2)
    {
        signal.isBullishRejection = true;
        signal.rejectionType = "ENGULFING";
    }
    else if(h1 <= h2 && l1 >= l2 && c1 > o1)
    {
        signal.isBullishRejection = true;
        signal.rejectionType = "INSIDE_BAR";
    }
    else if(c1 > o1 && o2 > c2 && c1 < o2 && o1 > c2)
    {
        signal.isBullishRejection = true;
        signal.rejectionType = "HARAMI";
    }
    
    if(upperWickRatio > 0.5 && bodyRatio < 0.4)
    {
        signal.isBearishRejection = true;
        signal.rejectionType = "SHOOTING_STAR";
    }
    else if(c1 < o1 && o2 < c2 && c1 < o2 && o1 <= c2)
    {
        signal.isBearishRejection = true;
        signal.rejectionType = "ENGULFING";
    }
    else if(h1 <= h2 && l1 >= l2 && c1 < o1)
    {
        signal.isBearishRejection = true;
        signal.rejectionType = "INSIDE_BAR";
    }
    else if(c1 < o1 && o2 < c2 && c1 > o2 && o1 < c2)
    {
        signal.isBearishRejection = true;
        signal.rejectionType = "HARAMI";
    }
    
    return signal;
}

//+------------------------------------------------------------------+
//| Calculate Dynamic Risk Based on Account State                    |
//+------------------------------------------------------------------+
double CalculateDynamicRiskPercentage()
{
    if(!UseAdvancedRisk)
        return RiskPercent;
    
    double currentBalance = AccountInfoDouble(ACCOUNT_BALANCE);
    
    if(currentBalance > g_advPeakBalance)
        g_advPeakBalance = currentBalance;
    
    if(g_advInitialBalance > 0)
        g_advCurrentProfit = ((currentBalance - g_advInitialBalance) / g_advInitialBalance) * 100.0;
    
    if(g_advPeakBalance > 0)
        g_advCurrentDrawdown = ((g_advPeakBalance - currentBalance) / g_advPeakBalance) * 100.0;
    
    double adjustedRisk = RiskPercent;
    
    if(UseDrawdownDefense && g_advCurrentDrawdown >= DrawdownThreshold)
    {
        adjustedRisk *= DrawdownRiskMultiplier;
        
        static datetime lastDrawdownAlert = 0;
        if(TimeCurrent() - lastDrawdownAlert > 3600)
        {
            PrintFormat("⚠️ DRAWDOWN DEFENSE: %.2f%% DD | Risk: %.2f%% → %.2f%%", 
                       g_advCurrentDrawdown, RiskPercent, adjustedRisk);
            lastDrawdownAlert = TimeCurrent();
        }
    }
    else if(UseProfitAccelerator && g_advCurrentProfit >= ProfitThreshold)
    {
        adjustedRisk *= ProfitRiskMultiplier;
        
        static datetime lastProfitAlert = 0;
        if(TimeCurrent() - lastProfitAlert > 3600)
        {
            PrintFormat("🚀 PROFIT ACCELERATOR: +%.2f%% | Risk: %.2f%% → %.2f%%", 
                       g_advCurrentProfit, RiskPercent, adjustedRisk);
            lastProfitAlert = TimeCurrent();
        }
    }
    
    adjustedRisk = MathMin(adjustedRisk, MaxRiskPercent);
    adjustedRisk = MathMax(adjustedRisk, 0.1);
    
    g_advAdjustedRisk = adjustedRisk;
    return adjustedRisk;
}

                       g_symbolLocks[i].GetStatus());
        }
    }
    
    if(ShowDashboard)
    {
        RemoveDashboard();
    }
    
    ChartRedraw();
    
    Print("========================================");
    Print("💾 Lock states saved to GlobalVariables");
    Print("🔄 Will be restored on next EA start");
    Print("========================================");
}

//+------------------------------------------------------------------+
//| Calculate Lot Size with Enhanced Risk Management                |
//+------------------------------------------------------------------+
double CalculateLotSize(const string symbol, const int index, const double atr)
{
    if(!g_eaSymbolInfo.Name(symbol) || atr <= 0)
        return g_eaSymbolInfo.LotsMin();
    
    double riskPercent = CalculateDynamicRiskPercentage();
    
    double accountBalance = AccountInfoDouble(ACCOUNT_BALANCE);
    double riskAmount = (riskPercent / 100.0) * accountBalance;
    
    double stopLossDistance = atr * ATRStopLoss;
    double tickValue = g_eaSymbolInfo.TickValue();
    double tickSize = g_eaSymbolInfo.TickSize();
    
    if(tickValue <= 0 || tickSize <= 0 || stopLossDistance <= 0)
        return g_eaSymbolInfo.LotsMin();
    
    double stopLossInTicks = stopLossDistance / tickSize;
    double lotSize = riskAmount / (stopLossInTicks * tickValue);
    
    if(symbol == "XAUUSD")
        lotSize /= 10.0;
    
    double minLot = g_eaSymbolInfo.LotsMin();
    double maxLot = g_eaSymbolInfo.LotsMax();
    double lotStep = g_eaSymbolInfo.LotsStep();
    
    lotSize = MathMax(minLot, MathMin(lotSize, maxLot));
    lotSize = NormalizeDouble(MathFloor(lotSize / lotStep) * lotStep, 2);
    
    if(UseAdvancedRisk && riskPercent != RiskPercent)
    {
        PrintFormat("%s: Risk %.2f%% (Base: %.2f%%) | Lot: %.2f | $%.2f at risk",
                   symbol, riskPercent, RiskPercent, lotSize, riskAmount);
    }
    
    return lotSize;
}

//+------------------------------------------------------------------+
//| Manage Positions                                                  |
//+------------------------------------------------------------------+
void ManagePositions(const string symbol, const int index)
{
    double atr = GetATRValue(index);
    if(atr <= 0)
        return;
    
    for(int i = PositionsTotal() - 1; i >= 0; i--)
    {
        if(!g_eaPosition.SelectByIndex(i))
            continue;
        
        if(g_eaPosition.Symbol() != symbol || g_eaPosition.Magic() != MagicNumber)
            continue;
        
        if(!g_eaSymbolInfo.Name(symbol))
            continue;
        
        g_eaSymbolInfo.RefreshRates();
        
        double currentSL = g_eaPosition.StopLoss();
        double currentTP = g_eaPosition.TakeProfit();
        double openPrice = g_eaPosition.PriceOpen();
        int digits = (int)g_eaSymbolInfo.Digits();
        
        if(currentTP > 0)
        {
            bool closeAtTP = false;
            double tpBuffer = g_eaSymbolInfo.Point() * 3;
            
            if(g_eaPosition.PositionType() == POSITION_TYPE_BUY)
            {
                double bid = g_eaSymbolInfo.Bid();
                
                if(bid >= (currentTP - tpBuffer))
                {
                    closeAtTP = true;
                    double pipsToTP = (currentTP - bid) / g_eaSymbolInfo.Point() / 10;
                    PrintFormat("%s BUY: Manual TP close triggered at %.5f (TP: %.5f, Distance: %.1f pips)", 
                               symbol, bid, currentTP, pipsToTP);
                }
            }
            else if(g_eaPosition.PositionType() == POSITION_TYPE_SELL)
            {
                double ask = g_eaSymbolInfo.Ask();
                
                if(ask <= (currentTP + tpBuffer))
                {
                    closeAtTP = true;
                    double pipsToTP = (ask - currentTP) / g_eaSymbolInfo.Point() / 10;
                    PrintFormat("%s SELL: Manual TP close triggered at %.5f (TP: %.5f, Distance: %.1f pips)", 
                               symbol, ask, currentTP, pipsToTP);
                }
            }
            
            if(closeAtTP)
            {
                ulong ticket = g_eaPosition.Ticket();
                double profit = g_eaPosition.Profit() + g_eaPosition.Swap() + g_eaPosition.Commission();
                
                if(g_eaTrade.PositionClose(ticket))
                {
                    if(g_eaTrade.ResultRetcode() == TRADE_RETCODE_DONE)
                    {
                        PrintFormat("✅ %s: Position #%I64u closed at TP | Profit: $%.2f", 
                                   symbol, ticket, profit);
                    }
                }
                continue;
            }
        }
        
        if(currentSL > 0)
        {
            bool closeAtSL = false;
            double slBuffer = g_eaSymbolInfo.Point() * 3;
            
            if(g_eaPosition.PositionType() == POSITION_TYPE_BUY)
            {
                double bid = g_eaSymbolInfo.Bid();
                
                if(bid <= (currentSL + slBuffer))
                {
                    closeAtSL = true;
                }
            }
            else if(g_eaPosition.PositionType() == POSITION_TYPE_SELL)
            {
                double ask = g_eaSymbolInfo.Ask();
                
                if(ask >= (currentSL - slBuffer))
                {
                    closeAtSL = true;
                }
            }
            
            if(closeAtSL)
            {
                ulong ticket = g_eaPosition.Ticket();
                
                if(g_eaTrade.PositionClose(ticket))
                {
                    if(g_eaTrade.ResultRetcode() == TRADE_RETCODE_DONE)
                    {
                        PrintFormat("🛑 %s: Position #%I64u closed at SL", symbol, ticket);
                    }
                }
                continue;
            }
        }
        
        bool modified = false;
        double newSL = 0.0;
        
        if(g_eaPosition.PositionType() == POSITION_TYPE_BUY)
        {
            double bid = g_eaSymbolInfo.Bid();
            double profitDistance = bid - openPrice;
            
            if(profitDistance >= (atr * BEMultiplier) && currentSL < openPrice)
            {
                newSL = NormalizeDouble(openPrice + (g_eaSymbolInfo.Point() * 10), digits);
                modified = true;
            }
            else if(profitDistance >= (atr * TrailStartMultiplier) && currentSL >= openPrice)
            {
                double trailSL = NormalizeDouble(bid - (atr * TrailStopMultiplier), digits);
                if(trailSL > currentSL)
                {
                    newSL = trailSL;
                    modified = true;
                }
            }
        }
        else if(g_eaPosition.PositionType() == POSITION_TYPE_SELL)
        {
            double ask = g_eaSymbolInfo.Ask();
            double profitDistance = openPrice - ask;
            
            if(profitDistance >= (atr * BEMultiplier) && (currentSL == 0 || currentSL > openPrice))
            {
                newSL = NormalizeDouble(openPrice - (g_eaSymbolInfo.Point() * 10), digits);
                modified = true;
            }
            else if(profitDistance >= (atr * TrailStartMultiplier) && currentSL > 0 && currentSL <= openPrice)
            {
                double trailSL = NormalizeDouble(ask + (atr * TrailStopMultiplier), digits);
                if(currentSL == 0 || trailSL < currentSL)
                {
                    newSL = trailSL;
                    modified = true;
                }
            }
        }
        
        if(modified && newSL > 0)
        {
            g_eaTrade.PositionModify(g_eaPosition.Ticket(), newSL, g_eaPosition.TakeProfit());
        }
    }
}

//+------------------------------------------------------------------+
//| Check Trading Session                                            |
//+------------------------------------------------------------------+
bool IsInTradingSession()
{
    MqlDateTime current;
    TimeToStruct(TimeCurrent(), current);
    int currentHour = current.hour;
    
    if(currentHour >= LondonStart && currentHour < LondonEnd)
        return true;
    
    if(currentHour >= NewYorkStart && currentHour < NewYorkEnd)
        return true;
    
    return false;
}

//+------------------------------------------------------------------+
//| Check Spread                                                     |
//+------------------------------------------------------------------+
bool IsSpreadAcceptable(const string symbol)
{
    if(!g_eaSymbolInfo.Name(symbol))
        return false;
    
    g_eaSymbolInfo.RefreshRates();
    
    double spread = g_eaSymbolInfo.Spread() * g_eaSymbolInfo.Point();
    double maxSpread = MaxSpreadPips * g_eaSymbolInfo.Point() * 10;
    
    if(spread > maxSpread)
    {
        return false;
    }
    
    return true;
}

//+------------------------------------------------------------------+
//| Update Statistics                                                |
//+------------------------------------------------------------------+
void UpdateStatistics()
{
    if(!HistorySelect(0, TimeCurrent()))
        return;
    
    for(int i = 0; i < SYMBOL_COUNT; i++)
    {
        if(!g_tradingSymbols[i].isActive)
            continue;
        
        string symbol = g_tradingSymbols[i].symbol;
        double totalProfit = 0;
        int totalTrades = 0;
        int wins = 0;
        int losses = 0;
        
        for(int j = HistoryDealsTotal() - 1; j >= 0; j--)
        {
            ulong ticket = HistoryDealGetTicket(j);
            if(ticket == 0)
                continue;
            
            if(HistoryDealGetString(ticket, DEAL_SYMBOL) != symbol)
                continue;
            
            if(HistoryDealGetInteger(ticket, DEAL_MAGIC) != MagicNumber)
                continue;
            
            if(HistoryDealGetInteger(ticket, DEAL_ENTRY) != DEAL_ENTRY_OUT)
                continue;
            
            double profit = HistoryDealGetDouble(ticket, DEAL_PROFIT);
            totalProfit += profit;
            totalTrades++;
            
            if(profit > 0)
                wins++;
            else if(profit < 0)
                losses++;
        }
        
        g_stats[i].totalProfit = totalProfit;
        g_stats[i].totalTrades = totalTrades;
        g_stats[i].winningTrades = wins;
        g_stats[i].losingTrades = losses;
    }
}

//| Update Dashboard                                                  |
//+------------------------------------------------------------------+
void UpdateDashboard()
{
    double balance = AccountInfoDouble(ACCOUNT_BALANCE);
    double equity = AccountInfoDouble(ACCOUNT_EQUITY);
    double profit = equity - balance;
    int totalPos = GetAllPositionsCount();
    
    string riskStatus = GetAdvancedRiskStatus();
    
    string accountText = StringFormat("Balance: $%.2f | Equity: $%.2f | P/L: $%.2f | Pos: %d/%d%s",
                                     balance, equity, profit, totalPos, MaxTotalPositions, riskStatus);
    ObjectSetString(0, "Dashboard_Account", OBJPROP_TEXT, accountText);
    ObjectSetInteger(0, "Dashboard_Account", OBJPROP_COLOR, profit >= 0 ? clrLime : clrRed);
    
    bool inSession = IsInTradingSession();
    string sessionText = StringFormat("Session: %s | HTF: D1 | LTF: H1 | 🔒 Locks: ON",
                                     inSession ? "ACTIVE" : "CLOSED");
    ObjectSetString(0, "Dashboard_Session", OBJPROP_TEXT, sessionText);
    ObjectSetInteger(0, "Dashboard_Session", OBJPROP_COLOR, inSession ? clrLime : clrOrange);
    
    if(UseAdvancedRisk)
    {
        string riskStateText = "";
        color riskColor = clrWhite;
        
        if(g_advCurrentDrawdown >= DrawdownThreshold && UseDrawdownDefense)
        {
            riskStateText = StringFormat("Risk: %.2f%% | ⚠️ DEFENSE (%.1f%% DD)", 
                                       g_advAdjustedRisk, g_advCurrentDrawdown);
            riskColor = clrOrange;
        }
        else if(g_advCurrentProfit >= ProfitThreshold && UseProfitAccelerator)
        {
            riskStateText = StringFormat("Risk: %.2f%% | 🚀 ACCELERATOR (+%.1f%%)", 
                                       g_advAdjustedRisk, g_advCurrentProfit);
            riskColor = clrLime;
        }
        else
        {
            riskStateText = StringFormat("Risk: %.2f%% | ✅ NORMAL", g_advAdjustedRisk);
            riskColor = clrWhite;
        }
        
        ObjectSetString(0, "Dashboard_RiskState", OBJPROP_TEXT, riskStateText);
        ObjectSetInteger(0, "Dashboard_RiskState", OBJPROP_COLOR, riskColor);
    }
    
    for(int i = 0; i < SYMBOL_COUNT; i++)
    {
        string labelName = "Dashboard_Symbol_" + IntegerToString(i);
        
        if(!g_tradingSymbols[i].isActive)
        {
            ObjectSetString(0, labelName, OBJPROP_TEXT, "");
            continue;
        }
        
        string symbol = g_tradingSymbols[i].symbol;
        string status = "-";
        color textColor = clrGray;
        
        int positions = GetSymbolPositionsCount(symbol);
        
        if(positions > 0)
        {
            double symbolProfit = 0;
            for(int j = 0; j < PositionsTotal(); j++)
            {
                if(g_eaPosition.SelectByIndex(j))
                {
                    if(g_eaPosition.Symbol() == symbol && g_eaPosition.Magic() == MagicNumber)
                    {
                        symbolProfit += g_eaPosition.Profit() + g_eaPosition.Swap() + g_eaPosition.Commission();
                    }
                }
            }
            
            status = StringFormat("IN TRADE: $%.2f", symbolProfit);
            textColor = symbolProfit >= 0 ? clrLime : clrRed;
        }
        else if(g_symbolLocks[i].isLocked)
        {
            string lockDate = TimeToString(g_symbolLocks[i].lockedD1Time, TIME_DATE);
            status = StringFormat("🔒 DONE (%s)", lockDate);
            textColor = clrOrange;
        }
        else
        {
            status = "✅ READY";
            textColor = clrGray;
        }
        
        string winRate = g_stats[i].totalTrades > 0 ? 
                        StringFormat("%.0f%%", (double)g_stats[i].winningTrades / g_stats[i].totalTrades * 100) : 
                        "-";
        
        string text = StringFormat("%s: %s | W/L: %d/%d (%s)", 
                                  symbol, status,
                                  g_stats[i].winningTrades, 
                                  g_stats[i].losingTrades,
                                  winRate);
        
        ObjectSetString(0, labelName, OBJPROP_TEXT, text);
        ObjectSetInteger(0, labelName, OBJPROP_COLOR, textColor);
    }
    
    ChartRedraw();
}

//+------------------------------------------------------------------+
//| OnTick                                                            |
//+------------------------------------------------------------------+
void OnTick()
{
    if(!g_initialized)
        return;
    
    datetime currentTime = TimeCurrent();
    if(currentTime - g_lastCheck < 60)
        return;
    
    g_lastCheck = currentTime;
    
    for(int i = 0; i < SYMBOL_COUNT; i++)
    {
        if(!g_tradingSymbols[i].isActive)
            continue;
        
        string symbol = g_tradingSymbols[i].symbol;
        
        CheckNewHTFCandle(symbol, i);
        ProcessSymbol(symbol, i);
    }
    
    if(ShowDashboard)
    {
        UpdateDashboard();
    }
}

//+------------------------------------------------------------------+
//| STEP 3: REPLACE ProcessSymbol() Function                         |
//+------------------------------------------------------------------+
void ProcessSymbol(const string symbol, const int index)
{
    // ========================================
    // 0️⃣ DAILY LOCK CHECK
    // ========================================
    if(!g_symbolLocks[index].CanTrade())
        return;
    
    g_symbolLocks[index].SetProcessing(true);
    
    // ========================================
    // 1️⃣ SESSION FILTER
    // ========================================
    if(UseSessionFilter && !IsInTradingSession())
    {
        g_symbolLocks[index].SetProcessing(false);
        return;
    }
    
    // ========================================
    // 2️⃣ MANAGE EXISTING POSITIONS
    // ========================================
    ManagePositions(symbol, index);
    
    // ========================================
    // 3️⃣ POSITION LIMITS
    // ========================================
    if(GetSymbolPositionsCount(symbol) > 0 || GetAllPositionsCount() >= MaxTotalPositions)
    {
        g_symbolLocks[index].SetProcessing(false);
        return;
    }
    
    // ========================================
    // 4️⃣ STEP 1: CHECK D1 BREAKOUT
    // ========================================
    if(!CheckD1Breakout(symbol, index))
    {
        g_symbolLocks[index].SetProcessing(false);
        return; // No D1 breakout yet
    }
    
    // ========================================
    // 5️⃣ STEP 2: CHECK H4 CONFIRMATION
    // ========================================
    if(!CheckH4Confirmation(symbol, index))
    {
        g_symbolLocks[index].SetProcessing(false);
        return; // H4 hasn't confirmed yet
    }
    
    // ========================================
    // 6️⃣ STEP 3: CHECK H1 PATTERN & SET ENTRY TIME
    // ========================================
    if(!CheckH1PatternAndSetEntry(symbol, index))
    {
        g_symbolLocks[index].SetProcessing(false);
        return; // Pattern not found yet
    }
    
    // ========================================
    // 7️⃣ STEP 4: CHECK IF READY TO ENTER (at H1 open)
    // ========================================
    if(!IsReadyToEnter(symbol, index))
    {
        g_symbolLocks[index].SetProcessing(false);
        return; // Not at entry H1 yet
    }
    
    // ========================================
    // 8️⃣ SPREAD FILTER (final check before entry)
    // ========================================
    if(UseSpreadFilter && !IsSpreadAcceptable(symbol))
    {
        PrintFormat("⚠️ %s: All conditions met but spread too high - LOCKING anyway", symbol);
        g_symbolLocks[index].Lock();
        g_htfData[index].d1BreakoutDetected = false;
        g_htfData[index].h4ConfirmationReceived = false;
        g_htfData[index].h1PatternConfirmed = false;
        g_symbolLocks[index].SetProcessing(false);
        return;
    }
    
    // ========================================
    // 9️⃣ LOCK SYMBOL (All 4 steps completed)
    // ========================================
    PrintFormat("🔒 %s: D1→H4→H1→H1Open all confirmed - LOCKING", symbol);
    g_symbolLocks[index].Lock();
    
    // ========================================
    // 🔟 EXECUTE TRADE
    // ========================================
    bool tradeExecuted = ExecuteTrade(symbol, index, g_htfData[index].pendingDirection, g_htfData[index].breakLevel);
    
    if(tradeExecuted)
    {
        PrintFormat("✅ %s: Trade executed successfully at H1 open", symbol);
    }
    else
    {
        PrintFormat("⚠️ %s: Trade failed but setup was valid - day consumed", symbol);
    }
    
    // ========================================
    // ⑪ RESET STATE FLAGS
    // ========================================
    g_htfData[index].d1BreakoutDetected = false;
    g_htfData[index].h4ConfirmationReceived = false;
    g_htfData[index].h1PatternConfirmed = false;
    
    g_symbolLocks[index].SetProcessing(false);
}

//+------------------------------------------------------------------+
//| STEP 2: REPLACE CheckNewHTFCandle()                              |
//+------------------------------------------------------------------+
void CheckNewHTFCandle(const string symbol, const int index)
{
    datetime currentHTF = iTime(symbol, PERIOD_D1, 0);
    
    if(currentHTF != g_htfData[index].time)
    {
        if(g_htfData[index].time > 0)
        {
            PrintFormat("📅 %s: NEW D1 CANDLE - Resetting all breakout states", symbol);
        }
        
        // CRITICAL: Reset ALL state flags for new day
        g_htfData[index].d1BreakoutDetected = false;
        g_htfData[index].h4ConfirmationReceived = false;
        g_htfData[index].h1PatternConfirmed = false;
        g_htfData[index].breakoutUp = false;
        g_htfData[index].breakoutDown = false;
        g_htfData[index].d1BreakoutTime = 0;
        g_htfData[index].h4ConfirmationTime = 0;
        g_htfData[index].h1PatternTime = 0;
        g_htfData[index].entryH1Time = 0;
        
        if(!MarkPreviousHTF(symbol, index))
        {
            PrintFormat("⚠️ %s: Failed to mark HTF", symbol);
            return;
        }
    }
}

//+------------------------------------------------------------------+
//| Mark Previous HTF                                                |
//+------------------------------------------------------------------+
bool MarkPreviousHTF(const string symbol, const int index)
{
    double open[], close[], high[], low[];
    datetime time[];
    
    if(CopyOpen(symbol, PERIOD_D1, 1, 1, open) <= 0 ||
       CopyClose(symbol, PERIOD_D1, 1, 1, close) <= 0 ||
       CopyHigh(symbol, PERIOD_D1, 1, 1, high) <= 0 ||
       CopyLow(symbol, PERIOD_D1, 1, 1, low) <= 0 ||
       CopyTime(symbol, PERIOD_D1, 0, 1, time) <= 0)
    {
        return false;
    }
    
    datetime newD1Time = time[0];
    
    if(g_htfData[index].time > 0 && g_htfData[index].time != newD1Time)
    {
        PrintFormat("📅 %s: NEW D1 CANDLE", symbol);
        PrintFormat("   Previous: %s", TimeToString(g_htfData[index].time, TIME_DATE));
        PrintFormat("   Current:  %s", TimeToString(newD1Time, TIME_DATE));
    }
    
    g_htfData[index].open = open[0];
    g_htfData[index].close = close[0];
    g_htfData[index].high = high[0];
    g_htfData[index].low = low[0];
    g_htfData[index].time = newD1Time;
    
    if(symbol == _Symbol)
    {
        DrawHTFLevels(symbol, index);
    }
    
    return true;
}

void DrawHTFLevels(const string symbol, const int index)
{
    datetime startTime = iTime(symbol, PERIOD_D1, 1);
    datetime endTime = TimeCurrent() + PeriodSeconds(PERIOD_D1) * 2;
    
    ObjectDelete(0, "HTF_High_" + symbol);

    ObjectSetInteger(0, "HTF_High_" + symbol, OBJPROP_COLOR, clrGreen);

    ObjectSetInteger(0, "HTF_Low_" + symbol, OBJPROP_COLOR, clrRed);
    ObjectSetInteger(0, "HTF_Low_" + symbol, OBJPROP_WIDTH, 2);
    ObjectSetInteger(0, "HTF_Low_" + symbol, OBJPROP_STYLE, STYLE_SOLID);

    ChartRedraw();
} 

//+------------------------------------------------------------------+
//| NEW FUNCTION 1: Check D1 Breakout (ONE-TIME TRIGGER)             |
//+------------------------------------------------------------------+
bool CheckD1Breakout(const string symbol, const int index)
{
    // If already detected breakout today, skip
    if(g_htfData[index].d1BreakoutDetected)
        return true; // Already detected, move to next step
    
    // Get CURRENT D1 close
    double closeD1[];
    if(CopyClose(symbol, PERIOD_D1, 0, 1, closeD1) <= 0)
        return false;
    
    double currentD1Close = closeD1[0];
    
    // Check BREAKOUT UP
    if(currentD1Close > g_htfData[index].high)
    {
        g_htfData[index].d1BreakoutDetected = true;
        g_htfData[index].d1BreakoutTime = TimeCurrent();
        g_htfData[index].breakoutUp = true;
        g_htfData[index].pendingDirection = ORDER_TYPE_BUY;
        g_htfData[index].breakLevel = g_htfData[index].high;
        
        Print("┌────────────────────────────────────────┐");
        PrintFormat("│  STEP 1/4: D1 BREAKOUT UP - %s", symbol);
        PrintFormat("│  D1 Close: %.5f", currentD1Close);
        PrintFormat("│  Prev D1 High: %.5f", g_htfData[index].high);
        PrintFormat("│  ⏳ Waiting for H4 to close above %.5f", g_htfData[index].high);
        Print("└────────────────────────────────────────┘");
        return true;
    }
    
    // Check BREAKOUT DOWN
    if(currentD1Close < g_htfData[index].low)
    {
        g_htfData[index].d1BreakoutDetected = true;
        g_htfData[index].d1BreakoutTime = TimeCurrent();
        g_htfData[index].breakoutDown = true;
        g_htfData[index].pendingDirection = ORDER_TYPE_SELL;
        g_htfData[index].breakLevel = g_htfData[index].low;
        
        Print("┌────────────────────────────────────────┐");
        PrintFormat("│  STEP 1/4: D1 BREAKOUT DOWN - %s", symbol);
        PrintFormat("│  D1 Close: %.5f", currentD1Close);
        PrintFormat("│  Prev D1 Low: %.5f", g_htfData[index].low);
        PrintFormat("│  ⏳ Waiting for H4 to close below %.5f", g_htfData[index].low);
        Print("└────────────────────────────────────────┘");
        return true;
    }
    
    return false; // No breakout yet
}

//+------------------------------------------------------------------+
//| NEW FUNCTION 2: Wait for H4 Close in Breakout Direction          |
//+------------------------------------------------------------------+
bool CheckH4Confirmation(const string symbol, const int index)
{
    // Must have D1 breakout first
    if(!g_htfData[index].d1BreakoutDetected)
        return false;
    
    // Already got H4 confirmation
    if(g_htfData[index].h4ConfirmationReceived)
        return true; // Already confirmed, move to next step
    
    // Get the LAST CLOSED H4 candle using iClose and iTime
    double h4Close = iClose(symbol, PERIOD_H4, 1);
    datetime h4Time = iTime(symbol, PERIOD_H4, 1);
    
    if(h4Close == 0 || h4Time == 0)
        return false;
    
    // Skip if we already checked this H4 candle
    if(h4Time <= g_htfData[index].h4ConfirmationTime)
        return false;
    
    // Update last checked time (even if no confirmation)
    g_htfData[index].h4ConfirmationTime = h4Time;
    
    // Check if this H4 candle closed AFTER the D1 breakout was detected
    if(h4Time < g_htfData[index].d1BreakoutTime)
        return false;
    
    // BULLISH: H4 must close above previous D1 high
    if(g_htfData[index].breakoutUp && h4Close > g_htfData[index].high)
    {
        g_htfData[index].h4ConfirmationReceived = true;
        
        Print("┌────────────────────────────────────────┐");
        PrintFormat("│  STEP 2/4: H4 CONFIRMATION - %s", symbol);
        PrintFormat("│  H4 Close: %.5f", h4Close);
        PrintFormat("│  Above: %.5f", g_htfData[index].high);
        PrintFormat("│  H4 Time: %s", TimeToString(h4Time, TIME_MINUTES));
        PrintFormat("│  ⏳ Waiting for H1 bullish pattern...");
        Print("└────────────────────────────────────────┘");
        return true;
    }
    
    // BEARISH: H4 must close below previous D1 low
    if(g_htfData[index].breakoutDown && h4Close < g_htfData[index].low)
    {
        g_htfData[index].h4ConfirmationReceived = true;
        
        Print("┌────────────────────────────────────────┐");
        PrintFormat("│  STEP 2/4: H4 CONFIRMATION - %s", symbol);
        PrintFormat("│  H4 Close: %.5f", h4Close);
        PrintFormat("│  Below: %.5f", g_htfData[index].low);
        PrintFormat("│  H4 Time: %s", TimeToString(h4Time, TIME_MINUTES));
        PrintFormat("│  ⏳ Waiting for H1 bearish pattern...");
        Print("└────────────────────────────────────────┘");
        return true;
    }
    
    return false; // H4 hasn't confirmed yet
}

//+------------------------------------------------------------------+
//| NEW FUNCTION 3: Check H1 Pattern & Set Entry Time                |
//+------------------------------------------------------------------+
bool CheckH1PatternAndSetEntry(const string symbol, const int index)
{
    // Must have both D1 breakout AND H4 confirmation
    if(!g_htfData[index].d1BreakoutDetected || !g_htfData[index].h4ConfirmationReceived)
        return false;
    
    // Already found pattern and set entry time
    if(g_htfData[index].h1PatternConfirmed)
        return true; // Pattern already confirmed, move to next step
    
    // Get current H1 time using iTime
    datetime currentH1Time = iTime(symbol, PERIOD_H1, 0);
    if(currentH1Time == 0)
        return false;
    
    // Skip if we already checked this H1 candle
    if(currentH1Time == g_htfData[index].h1PatternTime)
        return false;
    
    // Get H1 pattern from CLOSED candles (your PricePosition function)
    RejectionSignal signal = PricePosition(symbol, g_htfData[index].breakLevel);
    
    // Check ADX filter
    if(!CheckADXFilter(symbol, index))
    {
        g_htfData[index].h1PatternTime = currentH1Time; // Mark as checked
        return false;
    }
    
    bool patternFound = false;
    
    // For BUY: need bullish rejection
    if(g_htfData[index].pendingDirection == ORDER_TYPE_BUY && signal.isBullishRejection)
    {
        patternFound = true;
        
        Print("┌────────────────────────────────────────┐");
        PrintFormat("│  STEP 3/4: H1 PATTERN FOUND - %s", symbol);
        PrintFormat("│  Pattern: %s (Bullish)", signal.rejectionType);
        PrintFormat("│  ADX: PASSED");
        PrintFormat("│  Current H1: %s", TimeToString(currentH1Time, TIME_MINUTES));
        Print("└────────────────────────────────────────┘");
    }
    
    // For SELL: need bearish rejection
    if(g_htfData[index].pendingDirection == ORDER_TYPE_SELL && signal.isBearishRejection)
    {
        patternFound = true;
        
        Print("┌────────────────────────────────────────┐");
        PrintFormat("│  STEP 3/4: H1 PATTERN FOUND - %s", symbol);
        PrintFormat("│  Pattern: %s (Bearish)", signal.rejectionType);
        PrintFormat("│  ADX: PASSED");
        PrintFormat("│  Current H1: %s", TimeToString(currentH1Time, TIME_MINUTES));
        Print("└────────────────────────────────────────┘");
    }
    
    // Update checked time
    g_htfData[index].h1PatternTime = currentH1Time;
    
    if(patternFound)
    {
        g_htfData[index].h1PatternConfirmed = true;
        
        // Calculate NEXT H1 open time (entry time)
        g_htfData[index].entryH1Time = currentH1Time + PeriodSeconds(PERIOD_H1);
        
        PrintFormat("│  ⏰ Entry scheduled at NEXT H1 open: %s", 
                   TimeToString(g_htfData[index].entryH1Time, TIME_MINUTES));
        Print("└────────────────────────────────────────┘");
        return true;
    }
    
    return false;
}

//+------------------------------------------------------------------+
//| NEW FUNCTION 4: Check if Ready to Enter at H1 Open               |
//+------------------------------------------------------------------+
bool IsReadyToEnter(const string symbol, const int index)
{
    // All conditions must be met
    if(!g_htfData[index].d1BreakoutDetected || 
       !g_htfData[index].h4ConfirmationReceived ||
       !g_htfData[index].h1PatternConfirmed)
        return false;
    
    // Get current H1 open time using iTime instead of CopyTime
    datetime currentH1Time = iTime(symbol, PERIOD_H1, 0);
    if(currentH1Time == 0)
        return false;
    
    // Check if we've reached the scheduled entry time
    if(currentH1Time >= g_htfData[index].entryH1Time)
    {
        Print("┌────────────────────────────────────────┐");
        PrintFormat("│  STEP 4/4: READY TO ENTER - %s", symbol);
        PrintFormat("│  Current H1: %s", TimeToString(currentH1Time, TIME_MINUTES));
        PrintFormat("│  ✅ All conditions met - entering at H1 open");
        Print("└────────────────────────────────────────┘");
        return true;
    }
    
    // Still waiting for next H1
    static datetime lastLog[6]; // Adjust size if you have different number of symbols
    if(TimeCurrent() - lastLog[index] > 300) // Log every 5 minutes
    {
        PrintFormat("⏳ %s: Waiting for H1 open at %s (Current: %s)", 
                   symbol, 
                   TimeToString(g_htfData[index].entryH1Time, TIME_MINUTES),
                   TimeToString(currentH1Time, TIME_MINUTES));
        lastLog[index] = TimeCurrent();
    }
    
    return false;
}

//+------------------------------------------------------------------+
//| Execute Trade - Returns TRUE only if trade succeeds              |
//+------------------------------------------------------------------+
bool ExecuteTrade(const string symbol, const int index, const ENUM_ORDER_TYPE orderType, const double breakLevel)
{
    if(!g_eaSymbolInfo.Name(symbol))
    {
        PrintFormat("❌ %s: Failed to get symbol info", symbol);
        return false;
    }
    
    g_eaSymbolInfo.RefreshRates();
    
    // Get ATR for position sizing
    double atr = GetATRValue(index);
    if(atr <= 0)
    {
        PrintFormat("❌ %s: Invalid ATR value", symbol);
        return false;
    }
    
    // Calculate lot size
    double lotSize = CalculateLotSize(symbol, index, atr);
    if(lotSize < g_eaSymbolInfo.LotsMin())
    {
        PrintFormat("❌ %s: Lot size %.2f below minimum %.2f", 
                   symbol, lotSize, g_eaSymbolInfo.LotsMin());
        return false;
    }
    
    // Calculate entry price, SL, and TP
    double price, sl, tp;
    string direction;
    
    if(orderType == ORDER_TYPE_BUY)
    {
        direction = "BUY";
        price = g_eaSymbolInfo.Ask();
        sl = NormalizeDouble(price - (atr * ATRStopLoss), g_eaSymbolInfo.Digits());
        tp = NormalizeDouble(price + (atr * ATRTakeProfit), g_eaSymbolInfo.Digits());
    }
    else
    {
        direction = "SELL";
        price = g_eaSymbolInfo.Bid();
        sl = NormalizeDouble(price + (atr * ATRStopLoss), g_eaSymbolInfo.Digits());
        tp = NormalizeDouble(price - (atr * ATRTakeProfit), g_eaSymbolInfo.Digits());
    }
    
    // Create trade comment with date
    datetime currentD1 = iTime(symbol, PERIOD_D1, 0);
    string dateStr = TimeToString(currentD1, TIME_DATE);
    string tradeComment = StringFormat("BO_%s_%s", direction, dateStr);
    
    // Execute trade
    bool result = g_eaTrade.PositionOpen(symbol, orderType, lotSize, price, sl, tp, tradeComment);
    
    if(result && g_eaTrade.ResultRetcode() == TRADE_RETCODE_DONE)
    {
        ulong ticket = g_eaTrade.ResultOrder();
        double riskPips = MathAbs(price - sl) / g_eaSymbolInfo.Point() / 10;
        double rewardPips = MathAbs(tp - price) / g_eaSymbolInfo.Point() / 10;
        
        Print("┌────────────────────────────────────────┐");
        PrintFormat("│  ✅ TRADE EXECUTED: %s %s", symbol, direction);
        PrintFormat("│     Ticket: #%I64u", ticket);
        PrintFormat("│     Date: %s", dateStr);
        PrintFormat("│     Lot Size: %.2f", lotSize);
        PrintFormat("│     Entry: %.5f", price);
        PrintFormat("│     Stop Loss: %.5f (%.1f pips)", sl, riskPips);
        PrintFormat("│     Take Profit: %.5f (%.1f pips)", tp, rewardPips);
        PrintFormat("│     Risk/Reward: 1:%.1f", rewardPips/riskPips);
        PrintFormat("│     Comment: %s", tradeComment);
        Print("└────────────────────────────────────────┘");
        
        // Update statistics
        g_stats[index].totalTrades++;
        g_stats[index].lastTradeTime = TimeCurrent();
        
        return true; // SUCCESS - will trigger lock
    }
    else
    {
        PrintFormat("❌ TRADE FAILED: %s %s", symbol, direction);
        PrintFormat("   Retcode: %d (%s)", 
                   g_eaTrade.ResultRetcode(), 
                   g_eaTrade.ResultRetcodeDescription());
        
        return false; // FAILED - but will still trigger lock to prevent repeated attempts
    }
}
